# Lighthouse — Інтелектуальний планувальник завдань і звичок

Стек: C# / .NET 10, ASP.NET Core Web API, EF Core + SQL Server, .NET MAUI (Android), LinUCB (власна реалізація на C#).

## Структура рішення

| Проєкт | Призначення |
|---|---|
| `src/Lighthouse.Shared` | DTO, enum-и — спільні для API і мобільного клієнта |
| `src/Lighthouse.Core` | Доменні сутності та AI-алгоритм LinUCB (без залежностей від EF/ASP.NET) |
| `src/Lighthouse.Api` | REST API, EF Core (DbContext), сервіси, автентифікація JWT |
| `src/Lighthouse.Mobile` | MAUI-застосунок (створюється з шаблону Rider, див. нижче) |
| `tests/Lighthouse.Core.Tests` | Модульні тести |

## Перший запуск

1. Потрібно: .NET 10 SDK, Docker Desktop, Rider, (за бажанням) SSMS.
2. `./scripts/init.ps1` — створить `Lighthouse.sln`, підключить проєкти та встановить NuGet-пакети.
3. Скопіюйте `.env.example` у `.env` і змініть паролі.
4. `docker compose up -d db` — піднімає SQL Server (підключення з SSMS: `localhost,1433`, логін `sa`).
5. Створіть початкову міграцію:
   `dotnet ef migrations add InitialCreate --project src/Lighthouse.Api --output-dir Data/Migrations`
6. У `src/Lighthouse.Api/appsettings.Development.json` перевірте пароль `sa` у рядку підключення, а `Database:AutoMigrate` = `true` застосує міграцію при старті. Або вручну:
   `dotnet ef database update --project src/Lighthouse.Api`
7. Запустіть `Lighthouse.Api` (профіль `http`, порт 5080). Документація API: http://localhost:5080/scalar/v1
8. Тести: `dotnet test`

Емулятор Android звертається до API на комп'ютері за адресою `http://10.0.2.2:5080`.

## Мобільний проєкт

У Rider: New Solution → .NET MAUI App, назва `Lighthouse.Mobile`, розташування `src/`. Далі
`dotnet sln add src/Lighthouse.Mobile` і посилання на `Lighthouse.Shared`. Структуру папок і код додамо наступним кроком.

## Docker (повне розгортання)

`docker compose up -d --build` — SQL Server + API на http://localhost:8080 (міграції застосовуються автоматично).

using FluentValidation;
using Microsoft.AspNetCore.Mvc;

namespace Lighthouse.Api.Controllers;

[ApiController]
public abstract class ApiControllerBase : ControllerBase
{
    /// <summary>Повертає null, якщо модель коректна, інакше — готову відповідь 400 з переліком помилок.</summary>
    protected async Task<ActionResult?> ValidateAsync<T>(IValidator<T> validator, T model, CancellationToken ct)
    {
        var result = await validator.ValidateAsync(model, ct);
        return result.IsValid
            ? null
            : ValidationProblem(new ValidationProblemDetails(result.ToDictionary()));
    }
}

using FluentValidation;
using Lighthouse.Api.Services.Auth;
using Lighthouse.Shared.Contracts.Auth;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;

namespace Lighthouse.Api.Controllers;

[Route("api/auth")]
[AllowAnonymous]
public sealed class AuthController(IAuthService auth) : ApiControllerBase
{
    [HttpPost("register")]
    public async Task<ActionResult<AuthResponse>> Register(
        RegisterRequest request, [FromServices] IValidator<RegisterRequest> validator, CancellationToken ct)
    {
        if (await ValidateAsync(validator, request, ct) is { } invalid) return invalid;
        return Ok(await auth.RegisterAsync(request, ct));
    }

    [HttpPost("login")]
    public async Task<ActionResult<AuthResponse>> Login(
        LoginRequest request, [FromServices] IValidator<LoginRequest> validator, CancellationToken ct)
    {
        if (await ValidateAsync(validator, request, ct) is { } invalid) return invalid;
        return Ok(await auth.LoginAsync(request, ct));
    }

    [HttpPost("refresh")]
    public async Task<ActionResult<AuthResponse>> Refresh(
        RefreshRequest request, [FromServices] IValidator<RefreshRequest> validator, CancellationToken ct)
    {

{

    }

using Lighthouse.Api.Data;
using Lighthouse.Api.Infrastructure;
using Lighthouse.Core.Entities;
using Lighthouse.Shared.Contracts.Auth;
using Microsoft.EntityFrameworkCore;

namespace Lighthouse.Api.Services.Auth;

public sealed class AuthService(LighthouseDbContext db, ITokenService tokens) : IAuthService
{
    private const int MaxFailedAttempts = 5;                            // FR-8
    private static readonly TimeSpan LockoutDuration = TimeSpan.FromMinutes(15);

    public async Task<AuthResponse> RegisterAsync(RegisterRequest request, CancellationToken ct)
    {
        var email = NormalizeEmail(request.Email);

        if (await db.Users.AnyAsync(u => u.Email == email, ct))
        {
            throw new AppException(StatusCodes.Status409Conflict, "Email is already registered.");
        }

        var user = new User
        {
            Name = request.Name.Trim(),
            Email = email,
            PasswordHash = BCrypt.Net.BCrypt.HashPassword(request.Password)
            // TODO (FR-4): надіслати лист підтвердження та виставляти IsEmailConfirmed після переходу за посиланням.
        };
        user.NotificationSettings = new NotificationSettings { UserId = user.Id };

        db.Users.Add(user);
        return await IssueTokensAsync(user, ct); // користувач і refresh-токен зберігаються однією транзакцією
    }

    public async Task<AuthResponse> LoginAsync(LoginRequest request, CancellationToken ct)
    {
        const string invalidCredentials = "Invalid email or password.";
        var email = NormalizeEmail(request.Email);

        var user = await db.Users.FirstOrDefaultAsync(u => u.Email == email && !u.IsDeleted, ct);
        if (user is null || user.PasswordHash is null)
        {
            throw new AppException(StatusCodes.Status401Unauthorized, invalidCredentials);
        }

        if (user.LockoutEndUtc is { } lockoutEnd && lockoutEnd > DateTime.UtcNow)
        {
            throw new AppException(StatusCodes.Status423Locked,
                $"Account is locked until {lockoutEnd:u}. Try again later.");
        }

        if (!BCrypt.Net.BCrypt.Verify(request.Password, user.PasswordHash))
        {
            user.FailedLoginCount++;
            if (user.FailedLoginCount >= MaxFailedAttempts)
            {
                user.LockoutEndUtc = DateTime.UtcNow + LockoutDuration;
                user.FailedLoginCount = 0;
            }

            await db.SaveChangesAsync(ct);
            throw new AppException(StatusCodes.Status401Unauthorized, invalidCredentials);
        }

        user.FailedLoginCount = 0;
        user.LockoutEndUtc = null;
        return await IssueTokensAsync(user, ct);
    }

    public async Task<AuthResponse> RefreshAsync(string refreshToken, CancellationToken ct)
    {
        var hash = tokens.Hash(refreshToken);
        var stored = await db.RefreshTokens
            .Include(t => t.User)
            .FirstOrDefaultAsync(t => t.TokenHash == hash, ct);

        if (stored is null || stored.RevokedAtUtc is not null || stored.ExpiresAtUtc <= DateTime.UtcNow)
        {
            throw new AppException(StatusCodes.Status401Unauthorized, "Invalid refresh token.");
        }

        stored.RevokedAtUtc = DateTime.UtcNow; // ротація: старий токен більше не діє
        return await IssueTokensAsync(stored.User, ct);
    }

    public async Task LogoutAsync(string refreshToken, CancellationToken ct)
    {
        var hash = tokens.Hash(refreshToken);
        var stored = await db.RefreshTokens.FirstOrDefaultAsync(t => t.TokenHash == hash, ct);

        if (stored is not null && stored.RevokedAtUtc is null)
        {
            stored.RevokedAtUtc = DateTime.UtcNow; // FR-13
            await db.SaveChangesAsync(ct);
        }
    }

    private async Task<AuthResponse> IssueTokensAsync(User user, CancellationToken ct)
    {
        var (accessToken, accessExpiresAt) = tokens.CreateAccessToken(user);
        var refresh = tokens.CreateRefreshToken();

        db.RefreshTokens.Add(new RefreshToken
        {
            UserId = user.Id,
            TokenHash = refresh.Hash,
            ExpiresAtUtc = refresh.ExpiresAtUtc
        });

        await db.SaveChangesAsync(ct);

        return new AuthResponse(accessToken, refresh.Token, accessExpiresAt,
            new UserDto(user.Id, user.Name, user.Email));
    }

    private static string NormalizeEmail(string email) => email.Trim().ToLowerInvariant();
}

using System.Security.Claims;
using System.Security.Cryptography;
using System.Text;
using Lighthouse.Api.Infrastructure;
using Lighthouse.Api.Options;
using Lighthouse.Core.Entities;
using Microsoft.Extensions.Options;
using Microsoft.IdentityModel.JsonWebTokens;
using Microsoft.IdentityModel.Tokens;

namespace Lighthouse.Api.Services.Auth;

public sealed class TokenService(IOptions<JwtOptions> options) : ITokenService
{
    private readonly JwtOptions _options = options.Value;

    public (string Token, DateTime ExpiresAtUtc) CreateAccessToken(User user)
    {
        var expires = DateTime.UtcNow.AddMinutes(_options.AccessTokenMinutes);
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_options.Secret));

        var descriptor = new SecurityTokenDescriptor
        {
            Subject = new ClaimsIdentity(new List<Claim>
            {
                new(TokenClaims.Subject, user.Id.ToString()),
                new(TokenClaims.Email, user.Email),
                new(TokenClaims.Name, user.Name)
            }),
            Issuer = _options.Issuer,
            Audience = _options.Audience,
            Expires = expires,
            SigningCredentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha256)
        };

        return (new JsonWebTokenHandler().CreateToken(descriptor), expires);
    }

    public RefreshTokenInfo CreateRefreshToken()
    {
        var token = Convert.ToBase64String(RandomNumberGenerator.GetBytes(64));
        return new RefreshTokenInfo(token, Hash(token), DateTime.UtcNow.AddDays(_options.RefreshTokenDays));
    }

    public string Hash(string token) =>
        Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(token)));
}

namespace Lighthouse.Api.Services.Tasks;

public sealed class TaskService(LighthouseDbContext db) : ITaskService
{
    public async Task<IReadOnlyList<TaskDto>> GetAsync(Guid userId, DateTime? modifiedSinceUtc, CancellationToken ct)
    {
        var query = db.Tasks.AsNoTracking().Include(t => t.Subtasks).Where(t => t.UserId == userId);

        query = modifiedSinceUtc is { } since
            ? query.Where(t => t.ServerModifiedAtUtc > since)   // дельта: разом із «надгробками» видалених
            : query.Where(t => !t.IsDeleted);                   // перше завантаження

        var items = await query.OrderBy(t => t.ServerModifiedAtUtc).ToListAsync(ct);
        return items.Select(ToDto).ToList();
    }

    public async Task<TaskDto> UpsertAsync(Guid userId, Guid id, UpsertTaskRequest request, CancellationToken ct)
    {
        if (request.CategoryId is { } categoryId &&
            !await db.Categories.AnyAsync(c => c.Id == categoryId && c.UserId == userId && !c.IsDeleted, ct))
        {
            throw new AppException(StatusCodes.Status400BadRequest, "Category not found.");
        }

        var task = await db.Tasks.Include(t => t.Subtasks).FirstOrDefaultAsync(t => t.Id == id, ct);

        // Чужий запис маскуємо під «не існує», щоб не розкривати наявність ідентифікатора.
        if (task is not null && task.UserId != userId)
        {
            throw new AppException(StatusCodes.Status404NotFound, "Task not found.");
        }

        var now = DateTime.UtcNow;
        var editedAt = request.ClientUpdatedAtUtc > now ? now : request.ClientUpdatedAtUtc;

        if (task is null)
        {
            task = new TaskItem { Id = id, UserId = userId, CreatedAtUtc = now };
            db.Tasks.Add(task);
        }
        else if (task.UpdatedAtUtc > editedAt)
        {
            return ToDto(task); // «останній запис»: на сервері вже новіша правка — клієнт отримає актуальний стан
        }

        task.Title = request.Title.Trim();
        task.Description = request.Description;
        task.DeadlineUtc = request.DeadlineUtc;
        task.Priority = request.Priority;
        task.CategoryId = request.CategoryId;
        task.ReminderAtUtc = request.ReminderAtUtc;
        task.CompletedAtUtc = request.CompletedAtUtc;
        task.IsDeleted = false;
        task.UpdatedAtUtc = editedAt;
        task.ServerModifiedAtUtc = now;

        SyncSubtasks(task, request.Subtasks, now);

        await db.SaveChangesAsync(ct);
        return ToDto(task);
    }

    public async Task DeleteAsync(Guid userId, Guid id, CancellationToken ct)
    {
        var task = await db.Tasks.FirstOrDefaultAsync(t => t.Id == id, ct);
        if (task is null)
        {
            return; // видалення ідемпотентне
        }

        if (task.UserId != userId)
        {
            throw new AppException(StatusCodes.Status404NotFound, "Task not found.");
        }

        var now = DateTime.UtcNow;
        task.IsDeleted = true; // архів; фізичне видалення через 14 днів — окрема фонова задача (UC-09)
        task.UpdatedAtUtc = now;
        task.ServerModifiedAtUtc = now;

        await db.SaveChangesAsync(ct);
    }

    private void SyncSubtasks(TaskItem task, IReadOnlyList<SubtaskDto> incoming, DateTime now)
    {
        var incomingIds = incoming.Select(s => s.Id).ToHashSet();

        foreach (var removed in task.Subtasks.Where(s => !incomingIds.Contains(s.Id)).ToList())
        {
            db.Subtasks.Remove(removed);
        }

        for (var order = 0; order < incoming.Count; order++)
        {
            var dto = incoming[order];
            var existing = task.Subtasks.FirstOrDefault(s => s.Id == dto.Id);

            if (existing is null)
            {
                // Add() явно, щоб EF не вважав сутність із заданим ключем уже існуючою.
                db.Subtasks.Add(new Subtask
                {
                    Id = dto.Id,
                    TaskItemId = task.Id,
                    Title = dto.Title.Trim(),
                    IsDone = dto.IsDone,
                    SortOrder = order,
                    UpdatedAtUtc = now,
                    ServerModifiedAtUtc = now
                });
            }
            else
            {
                existing.Title = dto.Title.Trim();
                existing.IsDone = dto.IsDone;
                existing.SortOrder = order;
                existing.UpdatedAtUtc = now;
                existing.ServerModifiedAtUtc = now;
            }
        }
    }

    private static TaskDto ToDto(TaskItem t) => new(
        t.Id,
        t.Title,
        t.Description,
        t.DeadlineUtc,
        t.Priority,
        t.CategoryId,
        t.ReminderAtUtc,
        t.CompletedAtUtc,
        t.Subtasks.OrderBy(s => s.SortOrder).Select(s => new SubtaskDto(s.Id, s.Title, s.IsDone)).ToList(),
        t.UpdatedAtUtc,
        t.ServerModifiedAtUtc,
        t.IsDeleted);
}

namespace Lighthouse.Core.AI;

/// <summary>
/// LinUCB із спільними параметрами (Li et al., 2010): оцінка винагороди θᵀx плюс бонус за невизначеність
/// α·√(xᵀA⁻¹x). Замість інверсії матриці на кожному кроці підтримується A⁻¹ і оновлюється за формулою
/// Шермана–Моррісона, тому Update має складність O(d²), а розмірність d невелика (≈8–12 ознак).
/// </summary>
public sealed class LinUcbModel
{
    private readonly double[][] _aInv;
    private readonly double[] _b;

    public int Dimension { get; }

    /// <summary>Коефіцієнт дослідження: більше значення — більше «цікавості» до невивчених контекстів.</summary>
    public double Alpha { get; }

    public LinUcbModel(int dimension, double alpha = 1.0)
    {
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(dimension);
        ArgumentOutOfRangeException.ThrowIfNegative(alpha);

        Dimension = dimension;
        Alpha = alpha;
        _b = new double[dimension];
        _aInv = new double[dimension][];
        for (var i = 0; i < dimension; i++)
        {
            _aInv[i] = new double[dimension];
            _aInv[i][i] = 1.0; // A = I на старті
        }
    }

    private LinUcbModel(LinUcbSnapshot snapshot) : this(snapshot.Dimension, snapshot.Alpha)
    {
        if (snapshot.AInv.Length != Dimension
            || snapshot.B.Length != Dimension
            || snapshot.AInv.Any(row => row.Length != Dimension))
        {
            throw new ArgumentException("Snapshot dimensions are inconsistent.", nameof(snapshot));
        }

        for (var i = 0; i < Dimension; i++)
        {
            Array.Copy(snapshot.AInv[i], _aInv[i], Dimension);
        }

        Array.Copy(snapshot.B, _b, Dimension);
    }

    public static LinUcbModel FromSnapshot(LinUcbSnapshot snapshot) => new(snapshot);

    public LinUcbSnapshot ToSnapshot() =>
        new(Dimension, Alpha, _aInv.Select(row => (double[])row.Clone()).ToArray(), (double[])_b.Clone());

    /// <summary>Очікувана винагорода θᵀx (без бонусу за дослідження).</summary>
    public double ExpectedReward(double[] context)
    {
        EnsureDimension(context);
        var theta = MultiplyMatrixVector(_aInv, _b);
        return Dot(theta, context);
    }

    /// <summary>Невизначеність √(xᵀA⁻¹x): зменшується, коли модель бачила схожі контексти.</summary>
    public double Uncertainty(double[] context)
    {
        EnsureDimension(context);
        var ax = MultiplyMatrixVector(_aInv, context);
        return Math.Sqrt(Math.Max(0.0, Dot(context, ax)));
    }

    /// <summary>Верхня довірча межа (UCB) — значення, за яким ранжуються завдання.</summary>
    public double Score(double[] context) => ExpectedReward(context) + Alpha * Uncertainty(context);

    /// <summary>Повертає індекси контекстів у порядку спадання Score.</summary>
    public IReadOnlyList<int> RankDescending(IReadOnlyList<double[]> contexts) =>
        contexts
            .Select((context, index) => (Index: index, Score: Score(context)))
            .OrderByDescending(item => item.Score)
            .Select(item => item.Index)
            .ToList();

    /// <summary>Навчання на одному спостереженні: A⁻¹ ← A⁻¹ − (A⁻¹x)(A⁻¹x)ᵀ / (1 + xᵀA⁻¹x), b ← b + r·x.</summary>
    public void Update(double[] context, double reward)
    {
        EnsureDimension(context);

        var ax = MultiplyMatrixVector(_aInv, context);
        var denominator = 1.0 + Dot(context, ax);

        for (var i = 0; i < Dimension; i++)
        {
            for (var j = 0; j < Dimension; j++)
            {
                _aInv[i][j] -= ax[i] * ax[j] / denominator;
            }

            _b[i] += reward * context[i];
        }
    }

    private void EnsureDimension(double[] context)
    {
        ArgumentNullException.ThrowIfNull(context);
        if (context.Length != Dimension)
        {
            throw new ArgumentException(
                $"Context has {context.Length} features, expected {Dimension}.", nameof(context));
        }
    }

    private static double Dot(double[] left, double[] right)
    {
        var sum = 0.0;
        for (var i = 0; i < left.Length; i++)
        {
            sum += left[i] * right[i];
        }

        return sum;
    }

    private static double[] MultiplyMatrixVector(double[][] matrix, double[] vector)
    {
        var result = new double[vector.Length];
        for (var i = 0; i < matrix.Length; i++)
        {
            result[i] = Dot(matrix[i], vector);
        }

        return result;
    }
}

public sealed class StringToBoolConverter : IValueConverter
{

﻿using System.Net.Http.Headers;
using System.Net.Http.Json;
using Lighthouse.Mobile.Services.Auth;
using Lighthouse.Shared.Contracts.Auth;

namespace Lighthouse.Mobile.Services.Api;

public sealed class ApiClient(HttpClient http, ITokenStore tokenStore)
{
    public async Task<AuthResponse> RegisterAsync(string name, string email, string password, CancellationToken ct)
    {
        var response = await http.PostAsJsonAsync("api/auth/register", new RegisterRequest(name, email, password), ct);
        return await ReadAuthResponseAsync(response, ct);
    }

    public async Task<AuthResponse> LoginAsync(string email, string password, CancellationToken ct)
    {
        var response = await http.PostAsJsonAsync("api/auth/login", new LoginRequest(email, password), ct);
        return await ReadAuthResponseAsync(response, ct);
    }

    public async Task<HttpResponseMessage> SendAuthorizedAsync(
        Func<HttpRequestMessage> requestFactory, CancellationToken ct)
    {
        var (accessToken, _) = await tokenStore.GetAsync();
        var response = await SendWithTokenAsync(requestFactory(), accessToken, ct);

        if (response.StatusCode != System.Net.HttpStatusCode.Unauthorized)
        {
            return response;
        }

        var (_, refreshToken) = await tokenStore.GetAsync();
        if (string.IsNullOrEmpty(refreshToken))
        {
            return response;
        }

        var refreshResponse = await http.PostAsJsonAsync("api/auth/refresh", new RefreshRequest(refreshToken), ct);
        if (!refreshResponse.IsSuccessStatusCode)
        {
            await tokenStore.ClearAsync();
            return response;
        }

        var auth = await refreshResponse.Content.ReadFromJsonAsync<AuthResponse>(cancellationToken: ct)
                   ?? throw new InvalidOperationException("Empty refresh response.");
        await tokenStore.SaveAsync(auth.AccessToken, auth.RefreshToken);

        return await SendWithTokenAsync(requestFactory(), auth.AccessToken, ct);
    }

    private async Task<HttpResponseMessage> SendWithTokenAsync(HttpRequestMessage request, string? token, CancellationToken ct)
    {
        if (!string.IsNullOrEmpty(token))
        {
            request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        }

        return await http.SendAsync(request, ct);
    }

    private static async Task<AuthResponse> ReadAuthResponseAsync(HttpResponseMessage response, CancellationToken ct)
    {
        if (!response.IsSuccessStatusCode)
        {
            var problem = await response.Content.ReadAsStringAsync(ct);
            throw new ApiException(response.StatusCode, problem);
        }

        return await response.Content.ReadFromJsonAsync<AuthResponse>(cancellationToken: ct)
               ?? throw new InvalidOperationException("Empty auth response.");
    }
}

    : Exception($"API error {(int)statusCode}: {details}")
{
    public System.Net.HttpStatusCode StatusCode { get; } = statusCode;
}

﻿namespace Lighthouse.Mobile.Services.Auth;

public sealed class SecureTokenStore : ITokenStore
{
    private const string AccessKey = "access_token";
    private const string RefreshKey = "refresh_token";

    public Task SaveAsync(string accessToken, string refreshToken) => Task.WhenAll(
        SecureStorage.Default.SetAsync(AccessKey, accessToken),
        SecureStorage.Default.SetAsync(RefreshKey, refreshToken));

    public async Task<(string? AccessToken, string? RefreshToken)> GetAsync()
    {
        var access = await SecureStorage.Default.GetAsync(AccessKey);
        var refresh = await SecureStorage.Default.GetAsync(RefreshKey);
        return (access, refresh);
    }

    public Task ClearAsync()
    {
        SecureStorage.Default.Remove(AccessKey);
        SecureStorage.Default.Remove(RefreshKey);
        return Task.CompletedTask;
    }
}

namespace Lighthouse.Mobile.ViewModels;

public sealed partial class LoginViewModel(ApiClient api, ITokenStore tokenStore) : ObservableObject
{
    [ObservableProperty]
    private string _email = string.Empty;

    [ObservableProperty]
    private string _password = string.Empty;

    [ObservableProperty]
    private string? _errorMessage;

    [ObservableProperty]
    private bool _isBusy;

    [RelayCommand]
    private async Task LoginAsync()
    {
        if (IsBusy) return;

        ErrorMessage = null;
        IsBusy = true;
        try
        {
            var auth = await api.LoginAsync(Email.Trim(), Password, CancellationToken.None);
            await tokenStore.SaveAsync(auth.AccessToken, auth.RefreshToken);
            ErrorMessage = $"Успішний вхід: {auth.User.Name}"; // тимчасово, замінимо на навігацію пізніше
        }
        catch (ApiException ex) when (ex.StatusCode is System.Net.HttpStatusCode.Unauthorized)
        {
            ErrorMessage = "Невірний email або пароль.";
        }
        catch (ApiException ex) when (ex.StatusCode is System.Net.HttpStatusCode.Locked)
        {
            ErrorMessage = "Забагато спроб входу. Спробуйте пізніше.";
        }
        catch (Exception)
        {
            ErrorMessage = "Немає з'єднання з сервером. Перевірте інтернет і спробуйте ще раз.";
        }
        finally
        {
            IsBusy = false;
        }
    }

    [RelayCommand]
    private Task GoBackAsync() => Shell.Current.GoToAsync("..");
}

using Lighthouse.Core.AI;
using Xunit;

namespace Lighthouse.Core.Tests;

public class LinUcbModelTests
{
    [Fact]
    public void NewModel_ScoreEqualsAlphaTimesNormOfContext()
    {
        var model = new LinUcbModel(3, alpha: 1.0);

        var score = model.Score([1, 0, 0]);

        Assert.Equal(1.0, score, precision: 10);
    }

    [Fact]
    public void Update_WithPositiveReward_RaisesExpectedRewardOnlyForSeenContext()
    {
        var model = new LinUcbModel(3, alpha: 0.0);
        double[] seen = [1, 0, 0];
        double[] unseen = [0, 1, 0];

        for (var i = 0; i < 10; i++)
        {
            model.Update(seen, 1.0);
        }

        Assert.True(model.Score(seen) > 0.9);
        Assert.Equal(0.0, model.Score(unseen), precision: 10);
    }

    [Fact]
    public void Update_ReducesUncertaintyForSeenContext()
    {
        var model = new LinUcbModel(3, alpha: 1.0);
        double[] context = [1, 0, 0];
        var before = model.Uncertainty(context);

        for (var i = 0; i < 10; i++)
        {
            model.Update(context, 0.5);
        }

        Assert.True(model.Uncertainty(context) < before / 2);
    }

    [Fact]
    public void Snapshot_RoundTrip_PreservesScores()
    {
        var model = new LinUcbModel(3, alpha: 0.7);
        model.Update([1, 0.5, 0], 1.0);
        model.Update([0, 1, 0.2], 0.0);

        var restored = LinUcbModel.FromSnapshot(model.ToSnapshot());

        double[] probe = [0.3, 0.6, 0.9];
        Assert.Equal(model.Score(probe), restored.Score(probe), precision: 12);
    }

    [Fact]
    public void RankDescending_PutsBetterContextFirst()
    {
        var model = new LinUcbModel(2, alpha: 0.0);
        for (var i = 0; i < 5; i++)
        {
            model.Update([1, 0], 1.0);
            model.Update([0, 1], 0.0);
        }

        var order = model.RankDescending([[0, 1], [1, 0]]);

        Assert.Equal(new[] { 1, 0 }, order.ToArray());
    }

    [Fact]
    public void Score_WithWrongDimension_Throws()
    {
        var model = new LinUcbModel(3);

        Assert.Throws<ArgumentException>(() => model.Score([1, 2]));
    }
}

# Lighthouse — Інтелектуальний планувальник завдань і звичок

Стек: C# / .NET 10, ASP.NET Core Web API, EF Core + SQL Server, .NET MAUI (Android), LinUCB (власна реалізація на C#).

## Структура рішення

| Проєкт | Призначення |
|---|---|
| `src/Lighthouse.Shared` | DTO, enum-и — спільні для API і мобільного клієнта |
| `src/Lighthouse.Core` | Доменні сутності та AI-алгоритм LinUCB (без залежностей від EF/ASP.NET) |
| `src/Lighthouse.Api` | REST API, EF Core (DbContext), сервіси, автентифікація JWT |
| `src/Lighthouse.Mobile` | MAUI-застосунок (створюється з шаблону Rider, див. нижче) |
| `tests/Lighthouse.Core.Tests` | Модульні тести |

## Перший запуск

1. Потрібно: .NET 10 SDK, Docker Desktop, Rider, (за бажанням) SSMS.
2. `./scripts/init.ps1` — створить `Lighthouse.sln`, підключить проєкти та встановить NuGet-пакети.
3. Скопіюйте `.env.example` у `.env` і змініть паролі.
4. `docker compose up -d db` — піднімає SQL Server (підключення з SSMS: `localhost,1433`, логін `sa`).
5. Створіть початкову міграцію:
   `dotnet ef migrations add InitialCreate --project src/Lighthouse.Api --output-dir Data/Migrations`
6. У `src/Lighthouse.Api/appsettings.Development.json` перевірте пароль `sa` у рядку підключення, а `Database:AutoMigrate` = `true` застосує міграцію при старті. Або вручну:
   `dotnet ef database update --project src/Lighthouse.Api`
7. Запустіть `Lighthouse.Api` (профіль `http`, порт 5080). Документація API: http://localhost:5080/scalar/v1
8. Тести: `dotnet test`

Емулятор Android звертається до API на комп'ютері за адресою `http://10.0.2.2:5080`.

## Мобільний проєкт

У Rider: New Solution → .NET MAUI App, назва `Lighthouse.Mobile`, розташування `src/`. Далі
`dotnet sln add src/Lighthouse.Mobile` і посилання на `Lighthouse.Shared`. Структуру папок і код додамо наступним кроком.

## Docker (повне розгортання)

`docker compose up -d --build` — SQL Server + API на http://localhost:8080 (міграції застосовуються автоматично).

using FluentValidation;
using Microsoft.AspNetCore.Mvc;

namespace Lighthouse.Api.Controllers;

[ApiController]
public abstract class ApiControllerBase : ControllerBase
{
    /// <summary>Повертає null, якщо модель коректна, інакше — готову відповідь 400 з переліком помилок.</summary>
    protected async Task<ActionResult?> ValidateAsync<T>(IValidator<T> validator, T model, CancellationToken ct)
    {
        var result = await validator.ValidateAsync(model, ct);
        return result.IsValid
            ? null
            : ValidationProblem(new ValidationProblemDetails(result.ToDictionary()));
    }
}

using Lighthouse.Core.Entities;
using Microsoft.EntityFrameworkCore;

namespace Lighthouse.Api.Data;

public sealed class LighthouseDbContext(DbContextOptions<LighthouseDbContext> options) : DbContext(options)
{
    public DbSet<User> Users => Set<User>();
    public DbSet<RefreshToken> RefreshTokens => Set<RefreshToken>();
    public DbSet<NotificationSettings> NotificationSettings => Set<NotificationSettings>();
    public DbSet<Category> Categories => Set<Category>();
    public DbSet<TaskItem> Tasks => Set<TaskItem>();
    public DbSet<Subtask> Subtasks => Set<Subtask>();
    public DbSet<Habit> Habits => Set<Habit>();
    public DbSet<HabitLog> HabitLogs => Set<HabitLog>();
    public DbSet<BehaviorEvent> BehaviorEvents => Set<BehaviorEvent>();
    public DbSet<RecommendationFeedback> RecommendationFeedbacks => Set<RecommendationFeedback>();
    public DbSet<BanditState> BanditStates => Set<BanditState>();

    protected override void OnModelCreating(ModelBuilder b)
    {
        // Ідентифікатори генерує клієнт (офлайн-створення), тому БД їх не генерує.
        foreach (var entityType in b.Model.GetEntityTypes()
                     .Where(t => typeof(BaseEntity).IsAssignableFrom(t.ClrType)))
        {
            b.Entity(entityType.ClrType).Property(nameof(BaseEntity.Id)).ValueGeneratedNever();
        }

        b.Entity<User>(e =>
        {
            e.Property(x => x.Name).HasMaxLength(100).IsRequired();
            e.Property(x => x.Email).HasMaxLength(256).IsRequired();
            e.Property(x => x.PasswordHash).HasMaxLength(100);
            e.Property(x => x.GoogleSubject).HasMaxLength(64);
            e.Property(x => x.TimeZoneId).HasMaxLength(64);
            e.HasIndex(x => x.Email).IsUnique();
            e.HasOne(x => x.NotificationSettings)
                .WithOne()
                .HasForeignKey<NotificationSettings>(x => x.UserId)
                .OnDelete(DeleteBehavior.Cascade);
        });

        b.Entity<NotificationSettings>(e => e.HasKey(x => x.UserId));

        b.Entity<RefreshToken>(e =>
        {
            e.Property(x => x.TokenHash).HasMaxLength(64).IsRequired();
            e.HasIndex(x => x.TokenHash).IsUnique();
            e.HasOne(x => x.User)
                .WithMany(x => x.RefreshTokens)
                .HasForeignKey(x => x.UserId)
                .OnDelete(DeleteBehavior.Cascade);
        });

        b.Entity<Category>(e =>
        {
            e.Property(x => x.Name).HasMaxLength(100).IsRequired();
            e.Property(x => x.ColorHex).HasMaxLength(9);
            e.Property(x => x.Icon).HasMaxLength(50);
            e.HasIndex(x => new { x.UserId, x.ServerModifiedAtUtc });
            e.HasOne<User>().WithMany().HasForeignKey(x => x.UserId).OnDelete(DeleteBehavior.Cascade);
        });

        b.Entity<TaskItem>(e =>
        {
            e.ToTable("Tasks");
            e.Property(x => x.Title).HasMaxLength(200).IsRequired();
            e.Property(x => x.Description).HasMaxLength(2000);
            e.HasIndex(x => new { x.UserId, x.ServerModifiedAtUtc });
            e.HasOne<User>().WithMany().HasForeignKey(x => x.UserId).OnDelete(DeleteBehavior.Cascade);

            // NoAction: SQL Server забороняє кілька каскадних шляхів (User → Category → Task та User → Task).
            // Категорії видаляються «м'яко», а завдання переносяться в «Без категорії» у сервісі.
            e.HasOne<Category>().WithMany().HasForeignKey(x => x.CategoryId).OnDelete(DeleteBehavior.NoAction);
            e.HasMany(x => x.Subtasks).WithOne().HasForeignKey(x => x.TaskItemId).OnDelete(DeleteBehavior.Cascade);
        });

        b.Entity<Subtask>(e => e.Property(x => x.Title).HasMaxLength(200).IsRequired());

        b.Entity<Habit>(e =>
        {
            e.Property(x => x.Title).HasMaxLength(200).IsRequired();
            e.Property(x => x.Description).HasMaxLength(2000);
            e.Property(x => x.ColorHex).HasMaxLength(9);
            e.Property(x => x.Icon).HasMaxLength(50);
            e.HasIndex(x => new { x.UserId, x.ServerModifiedAtUtc });
            e.HasOne<User>().WithMany().HasForeignKey(x => x.UserId).OnDelete(DeleteBehavior.Cascade);
            e.HasOne<Category>().WithMany().HasForeignKey(x => x.CategoryId).OnDelete(DeleteBehavior.NoAction);
            e.HasMany(x => x.Logs).WithOne().HasForeignKey(x => x.HabitId).OnDelete(DeleteBehavior.Cascade);
        });

        b.Entity<HabitLog>(e => e.HasIndex(x => new { x.HabitId, x.Date }).IsUnique());

        b.Entity<BehaviorEvent>(e =>
        {
            e.Property(x => x.PayloadJson).HasColumnType("nvarchar(max)");
            e.HasIndex(x => new { x.UserId, x.OccurredAtUtc });
            e.HasOne<User>().WithMany().HasForeignKey(x => x.UserId).OnDelete(DeleteBehavior.Cascade);
        });

        b.Entity<RecommendationFeedback>(e =>
        {
            e.Property(x => x.RecommendationKey).HasMaxLength(100).IsRequired();
            e.Property(x => x.ContextJson).HasColumnType("nvarchar(max)");
            e.HasIndex(x => new { x.UserId, x.CreatedAtUtc });
            e.HasOne<User>().WithMany().HasForeignKey(x => x.UserId).OnDelete(DeleteBehavior.Cascade);
        });

        b.Entity<BanditState>(e =>
        {
            e.Property(x => x.SnapshotJson).HasColumnType("nvarchar(max)").IsRequired();
            e.HasIndex(x => x.UserId).IsUnique();
            e.HasOne<User>().WithMany().HasForeignKey(x => x.UserId).OnDelete(DeleteBehavior.Cascade);
        });
    }
}

using Lighthouse.Api.Data;
using Lighthouse.Api.Infrastructure;
using Lighthouse.Core.Entities;
using Lighthouse.Shared.Contracts.Auth;
using Microsoft.EntityFrameworkCore;

namespace Lighthouse.Api.Services.Auth;

public sealed class AuthService(LighthouseDbContext db, ITokenService tokens) : IAuthService
{
    private const int MaxFailedAttempts = 5;                            // FR-8
    private static readonly TimeSpan LockoutDuration = TimeSpan.FromMinutes(15);

    public async Task<AuthResponse> RegisterAsync(RegisterRequest request, CancellationToken ct)
    {
        var email = NormalizeEmail(request.Email);

        if (await db.Users.AnyAsync(u => u.Email == email, ct))
        {
            throw new AppException(StatusCodes.Status409Conflict, "Email is already registered.");
        }

        var user = new User
        {
            Name = request.Name.Trim(),
            Email = email,
            PasswordHash = BCrypt.Net.BCrypt.HashPassword(request.Password)
            // TODO (FR-4): надіслати лист підтвердження та виставляти IsEmailConfirmed після переходу за посиланням.
        };
        user.NotificationSettings = new NotificationSettings { UserId = user.Id };

        db.Users.Add(user);
        return await IssueTokensAsync(user, ct); // користувач і refresh-токен зберігаються однією транзакцією
    }

    public async Task<AuthResponse> LoginAsync(LoginRequest request, CancellationToken ct)
    {
        const string invalidCredentials = "Invalid email or password.";
        var email = NormalizeEmail(request.Email);

        var user = await db.Users.FirstOrDefaultAsync(u => u.Email == email && !u.IsDeleted, ct);
        if (user is null || user.PasswordHash is null)
        {
            throw new AppException(StatusCodes.Status401Unauthorized, invalidCredentials);
        }

        if (user.LockoutEndUtc is { } lockoutEnd && lockoutEnd > DateTime.UtcNow)
        {
            throw new AppException(StatusCodes.Status423Locked,
                $"Account is locked until {lockoutEnd:u}. Try again later.");
        }

        if (!BCrypt.Net.BCrypt.Verify(request.Password, user.PasswordHash))
        {
            user.FailedLoginCount++;
            if (user.FailedLoginCount >= MaxFailedAttempts)
            {
                user.LockoutEndUtc = DateTime.UtcNow + LockoutDuration;
                user.FailedLoginCount = 0;
            }

            await db.SaveChangesAsync(ct);
            throw new AppException(StatusCodes.Status401Unauthorized, invalidCredentials);
        }

        user.FailedLoginCount = 0;
        user.LockoutEndUtc = null;
        return await IssueTokensAsync(user, ct);
    }

    public async Task<AuthResponse> RefreshAsync(string refreshToken, CancellationToken ct)
    {
        var hash = tokens.Hash(refreshToken);
        var stored = await db.RefreshTokens
            .Include(t => t.User)
            .FirstOrDefaultAsync(t => t.TokenHash == hash, ct);

        if (stored is null || stored.RevokedAtUtc is not null || stored.ExpiresAtUtc <= DateTime.UtcNow)
        {
            throw new AppException(StatusCodes.Status401Unauthorized, "Invalid refresh token.");
        }

        stored.RevokedAtUtc = DateTime.UtcNow; // ротація: старий токен більше не діє
        return await IssueTokensAsync(stored.User, ct);
    }

    public async Task LogoutAsync(string refreshToken, CancellationToken ct)
    {
        var hash = tokens.Hash(refreshToken);
        var stored = await db.RefreshTokens.FirstOrDefaultAsync(t => t.TokenHash == hash, ct);

        if (stored is not null && stored.RevokedAtUtc is null)
        {
            stored.RevokedAtUtc = DateTime.UtcNow; // FR-13
            await db.SaveChangesAsync(ct);
        }
    }

    private async Task<AuthResponse> IssueTokensAsync(User user, CancellationToken ct)
    {
        var (accessToken, accessExpiresAt) = tokens.CreateAccessToken(user);
        var refresh = tokens.CreateRefreshToken();

        db.RefreshTokens.Add(new RefreshToken
        {
            UserId = user.Id,
            TokenHash = refresh.Hash,
            ExpiresAtUtc = refresh.ExpiresAtUtc
        });

        await db.SaveChangesAsync(ct);

        return new AuthResponse(accessToken, refresh.Token, accessExpiresAt,
            new UserDto(user.Id, user.Name, user.Email));
    }

    private static string NormalizeEmail(string email) => email.Trim().ToLowerInvariant();
}

using System.Security.Claims;
using System.Security.Cryptography;
using System.Text;
using Lighthouse.Api.Infrastructure;
using Lighthouse.Api.Options;
using Lighthouse.Core.Entities;
using Microsoft.Extensions.Options;
using Microsoft.IdentityModel.JsonWebTokens;
using Microsoft.IdentityModel.Tokens;

namespace Lighthouse.Api.Services.Auth;

public sealed class TokenService(IOptions<JwtOptions> options) : ITokenService
{
    private readonly JwtOptions _options = options.Value;

    public (string Token, DateTime ExpiresAtUtc) CreateAccessToken(User user)
    {
        var expires = DateTime.UtcNow.AddMinutes(_options.AccessTokenMinutes);
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_options.Secret));

        var descriptor = new SecurityTokenDescriptor
        {
            Subject = new ClaimsIdentity(new List<Claim>
            {
                new(TokenClaims.Subject, user.Id.ToString()),
                new(TokenClaims.Email, user.Email),
                new(TokenClaims.Name, user.Name)
            }),
            Issuer = _options.Issuer,
            Audience = _options.Audience,
            Expires = expires,
            SigningCredentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha256)
        };

        return (new JsonWebTokenHandler().CreateToken(descriptor), expires);
    }

    public RefreshTokenInfo CreateRefreshToken()
    {
        var token = Convert.ToBase64String(RandomNumberGenerator.GetBytes(64));
        return new RefreshTokenInfo(token, Hash(token), DateTime.UtcNow.AddDays(_options.RefreshTokenDays));
    }

    public string Hash(string token) =>
        Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(token)));
}

using Lighthouse.Api.Data;
using Lighthouse.Api.Infrastructure;
using Lighthouse.Core.Entities;
using Lighthouse.Shared.Contracts.Tasks;
using Microsoft.EntityFrameworkCore;

namespace Lighthouse.Api.Services.Tasks;

public sealed class TaskService(LighthouseDbContext db) : ITaskService
{
    public async Task<IReadOnlyList<TaskDto>> GetAsync(Guid userId, DateTime? modifiedSinceUtc, CancellationToken ct)
    {
        var query = db.Tasks.AsNoTracking().Include(t => t.Subtasks).Where(t => t.UserId == userId);

        query = modifiedSinceUtc is { } since
            ? query.Where(t => t.ServerModifiedAtUtc > since)   // дельта: разом із «надгробками» видалених
            : query.Where(t => !t.IsDeleted);                   // перше завантаження

        var items = await query.OrderBy(t => t.ServerModifiedAtUtc).ToListAsync(ct);
        return items.Select(ToDto).ToList();
    }

    public async Task<TaskDto> UpsertAsync(Guid userId, Guid id, UpsertTaskRequest request, CancellationToken ct)
    {
        if (request.CategoryId is { } categoryId &&
            !await db.Categories.AnyAsync(c => c.Id == categoryId && c.UserId == userId && !c.IsDeleted, ct))
        {
            throw new AppException(StatusCodes.Status400BadRequest, "Category not found.");
        }

        var task = await db.Tasks.Include(t => t.Subtasks).FirstOrDefaultAsync(t => t.Id == id, ct);

        // Чужий запис маскуємо під «не існує», щоб не розкривати наявність ідентифікатора.
        if (task is not null && task.UserId != userId)
        {
            throw new AppException(StatusCodes.Status404NotFound, "Task not found.");
        }

        var now = DateTime.UtcNow;
        var editedAt = request.ClientUpdatedAtUtc > now ? now : request.ClientUpdatedAtUtc;

        if (task is null)
        {
            task = new TaskItem { Id = id, UserId = userId, CreatedAtUtc = now };
            db.Tasks.Add(task);
        }
        else if (task.UpdatedAtUtc > editedAt)
        {
            return ToDto(task); // «останній запис»: на сервері вже новіша правка — клієнт отримає актуальний стан
        }

        task.Title = request.Title.Trim();
        task.Description = request.Description;
        task.DeadlineUtc = request.DeadlineUtc;
        task.Priority = request.Priority;
        task.CategoryId = request.CategoryId;
        task.ReminderAtUtc = request.ReminderAtUtc;
        task.CompletedAtUtc = request.CompletedAtUtc;
        task.IsDeleted = false;
        task.UpdatedAtUtc = editedAt;
        task.ServerModifiedAtUtc = now;

        SyncSubtasks(task, request.Subtasks, now);

        await db.SaveChangesAsync(ct);
        return ToDto(task);
    }

    public async Task DeleteAsync(Guid userId, Guid id, CancellationToken ct)
    {
        var task = await db.Tasks.FirstOrDefaultAsync(t => t.Id == id, ct);
        if (task is null)
        {
            return; // видалення ідемпотентне
        }

        if (task.UserId != userId)
        {
            throw new AppException(StatusCodes.Status404NotFound, "Task not found.");
        }

        var now = DateTime.UtcNow;
        task.IsDeleted = true; // архів; фізичне видалення через 14 днів — окрема фонова задача (UC-09)
        task.UpdatedAtUtc = now;
        task.ServerModifiedAtUtc = now;

        await db.SaveChangesAsync(ct);
    }

    private void SyncSubtasks(TaskItem task, IReadOnlyList<SubtaskDto> incoming, DateTime now)
    {
        var incomingIds = incoming.Select(s => s.Id).ToHashSet();

        foreach (var removed in task.Subtasks.Where(s => !incomingIds.Contains(s.Id)).ToList())
        {
            db.Subtasks.Remove(removed);
        }

        for (var order = 0; order < incoming.Count; order++)
        {
            var dto = incoming[order];
            var existing = task.Subtasks.FirstOrDefault(s => s.Id == dto.Id);

            if (existing is null)
            {
                // Add() явно, щоб EF не вважав сутність із заданим ключем уже існуючою.
                db.Subtasks.Add(new Subtask
                {
                    Id = dto.Id,
                    TaskItemId = task.Id,
                    Title = dto.Title.Trim(),
                    IsDone = dto.IsDone,
                    SortOrder = order,
                    UpdatedAtUtc = now,
                    ServerModifiedAtUtc = now
                });
            }
            else
            {
                existing.Title = dto.Title.Trim();
                existing.IsDone = dto.IsDone;
                existing.SortOrder = order;
                existing.UpdatedAtUtc = now;
                existing.ServerModifiedAtUtc = now;
            }
        }
    }

    private static TaskDto ToDto(TaskItem t) => new(
        t.Id,
        t.Title,
        t.Description,
        t.DeadlineUtc,
        t.Priority,
        t.CategoryId,
        t.ReminderAtUtc,
        t.CompletedAtUtc,
        t.Subtasks.OrderBy(s => s.SortOrder).Select(s => new SubtaskDto(s.Id, s.Title, s.IsDone)).ToList(),
        t.UpdatedAtUtc,
        t.ServerModifiedAtUtc,
        t.IsDeleted);
}

/// <summary>
/// LinUCB із спільними параметрами (Li et al., 2010): оцінка винагороди θᵀx плюс бонус за невизначеність
/// α·√(xᵀA⁻¹x). Замість інверсії матриці на кожному кроці підтримується A⁻¹ і оновлюється за формулою
/// Шермана–Моррісона, тому Update має складність O(d²), а розмірність d невелика (≈8–12 ознак).
/// </summary>
public sealed class LinUcbModel
{
    private readonly double[][] _aInv;
    private readonly double[] _b;

    public int Dimension { get; }

    /// <summary>Коефіцієнт дослідження: більше значення — більше «цікавості» до невивчених контекстів.</summary>
    public double Alpha { get; }

    public LinUcbModel(int dimension, double alpha = 1.0)
    {
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(dimension);
        ArgumentOutOfRangeException.ThrowIfNegative(alpha);

        Dimension = dimension;
        Alpha = alpha;
        _b = new double[dimension];
        _aInv = new double[dimension][];
        for (var i = 0; i < dimension; i++)
        {
            _aInv[i] = new double[dimension];
            _aInv[i][i] = 1.0; // A = I на старті
        }
    }

    private LinUcbModel(LinUcbSnapshot snapshot) : this(snapshot.Dimension, snapshot.Alpha)
    {
        if (snapshot.AInv.Length != Dimension
            || snapshot.B.Length != Dimension
            || snapshot.AInv.Any(row => row.Length != Dimension))
        {
            throw new ArgumentException("Snapshot dimensions are inconsistent.", nameof(snapshot));
        }

        for (var i = 0; i < Dimension; i++)
        {
            Array.Copy(snapshot.AInv[i], _aInv[i], Dimension);
        }

        Array.Copy(snapshot.B, _b, Dimension);
    }

    public static LinUcbModel FromSnapshot(LinUcbSnapshot snapshot) => new(snapshot);

    public LinUcbSnapshot ToSnapshot() =>
        new(Dimension, Alpha, _aInv.Select(row => (double[])row.Clone()).ToArray(), (double[])_b.Clone());

    /// <summary>Очікувана винагорода θᵀx (без бонусу за дослідження).</summary>
    public double ExpectedReward(double[] context)
    {
        EnsureDimension(context);
        var theta = MultiplyMatrixVector(_aInv, _b);
        return Dot(theta, context);
    }

    /// <summary>Невизначеність √(xᵀA⁻¹x): зменшується, коли модель бачила схожі контексти.</summary>
    public double Uncertainty(double[] context)
    {
        EnsureDimension(context);
        var ax = MultiplyMatrixVector(_aInv, context);
        return Math.Sqrt(Math.Max(0.0, Dot(context, ax)));
    }

    /// <summary>Верхня довірча межа (UCB) — значення, за яким ранжуються завдання.</summary>
    public double Score(double[] context) => ExpectedReward(context) + Alpha * Uncertainty(context);

    /// <summary>Повертає індекси контекстів у порядку спадання Score.</summary>
    public IReadOnlyList<int> RankDescending(IReadOnlyList<double[]> contexts) =>
        contexts
            .Select((context, index) => (Index: index, Score: Score(context)))
            .OrderByDescending(item => item.Score)
            .Select(item => item.Index)
            .ToList();

    /// <summary>Навчання на одному спостереженні: A⁻¹ ← A⁻¹ − (A⁻¹x)(A⁻¹x)ᵀ / (1 + xᵀA⁻¹x), b ← b + r·x.</summary>
    public void Update(double[] context, double reward)
    {
        EnsureDimension(context);

        var ax = MultiplyMatrixVector(_aInv, context);
        var denominator = 1.0 + Dot(context, ax);

        for (var i = 0; i < Dimension; i++)
        {
            for (var j = 0; j < Dimension; j++)
            {
                _aInv[i][j] -= ax[i] * ax[j] / denominator;
            }

            _b[i] += reward * context[i];
        }
    }

    private void EnsureDimension(double[] context)
    {
        ArgumentNullException.ThrowIfNull(context);
        if (context.Length != Dimension)
        {
            throw new ArgumentException(
                $"Context has {context.Length} features, expected {Dimension}.", nameof(context));
        }
    }

    private static double Dot(double[] left, double[] right)
    {
        var sum = 0.0;
        for (var i = 0; i < left.Length; i++)
        {
            sum += left[i] * right[i];
        }

        return sum;
    }

    private static double[] MultiplyMatrixVector(double[][] matrix, double[] vector)
    {
        var result = new double[vector.Length];
        for (var i = 0; i < matrix.Length; i++)
        {
            result[i] = Dot(matrix[i], vector);
        }

        return result;
    }
}

﻿using System.Net.Http.Headers;
using System.Net.Http.Json;
using Lighthouse.Mobile.Services.Auth;
using Lighthouse.Shared.Contracts.Auth;

namespace Lighthouse.Mobile.Services.Api;

public sealed class ApiClient(HttpClient http, ITokenStore tokenStore)
{
    public async Task<AuthResponse> RegisterAsync(string name, string email, string password, CancellationToken ct)
    {
        var response = await http.PostAsJsonAsync("api/auth/register", new RegisterRequest(name, email, password), ct);
        return await ReadAuthResponseAsync(response, ct);
    }

    public async Task<AuthResponse> LoginAsync(string email, string password, CancellationToken ct)
    {
        var response = await http.PostAsJsonAsync("api/auth/login", new LoginRequest(email, password), ct);
        return await ReadAuthResponseAsync(response, ct);
    }

    public async Task<HttpResponseMessage> SendAuthorizedAsync(
        Func<HttpRequestMessage> requestFactory, CancellationToken ct)
    {
        var (accessToken, _) = await tokenStore.GetAsync();
        var response = await SendWithTokenAsync(requestFactory(), accessToken, ct);

        if (response.StatusCode != System.Net.HttpStatusCode.Unauthorized)
        {
            return response;
        }

        var (_, refreshToken) = await tokenStore.GetAsync();
        if (string.IsNullOrEmpty(refreshToken))
        {
            return response;
        }

        var refreshResponse = await http.PostAsJsonAsync("api/auth/refresh", new RefreshRequest(refreshToken), ct);
        if (!refreshResponse.IsSuccessStatusCode)
        {
            await tokenStore.ClearAsync();
            return response;
        }

        var auth = await refreshResponse.Content.ReadFromJsonAsync<AuthResponse>(cancellationToken: ct)
                   ?? throw new InvalidOperationException("Empty refresh response.");
        await tokenStore.SaveAsync(auth.AccessToken, auth.RefreshToken);

        return await SendWithTokenAsync(requestFactory(), auth.AccessToken, ct);
    }

    private async Task<HttpResponseMessage> SendWithTokenAsync(HttpRequestMessage request, string? token, CancellationToken ct)
    {
        if (!string.IsNullOrEmpty(token))
        {
            request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        }

        return await http.SendAsync(request, ct);
    }

    private static async Task<AuthResponse> ReadAuthResponseAsync(HttpResponseMessage response, CancellationToken ct)
    {
        if (!response.IsSuccessStatusCode)
        {
            var problem = await response.Content.ReadAsStringAsync(ct);
            throw new ApiException(response.StatusCode, problem);
        }

        return await response.Content.ReadFromJsonAsync<AuthResponse>(cancellationToken: ct)
               ?? throw new InvalidOperationException("Empty auth response.");
    }
}

public sealed class ApiException(System.Net.HttpStatusCode statusCode, string details)
    : Exception($"API error {(int)statusCode}: {details}")
{
    public System.Net.HttpStatusCode StatusCode { get; } = statusCode;
}

﻿namespace Lighthouse.Mobile.Services.Auth;

public sealed class SecureTokenStore : ITokenStore
{
    private const string AccessKey = "access_token";
    private const string RefreshKey = "refresh_token";

    public Task SaveAsync(string accessToken, string refreshToken) => Task.WhenAll(
        SecureStorage.Default.SetAsync(AccessKey, accessToken),
        SecureStorage.Default.SetAsync(RefreshKey, refreshToken));

    public async Task<(string? AccessToken, string? RefreshToken)> GetAsync()
    {
        var access = await SecureStorage.Default.GetAsync(AccessKey);
        var refresh = await SecureStorage.Default.GetAsync(RefreshKey);
        return (access, refresh);
    }

    public Task ClearAsync()
    {
        SecureStorage.Default.Remove(AccessKey);
        SecureStorage.Default.Remove(RefreshKey);
        return Task.CompletedTask;
    }
}

﻿using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using Lighthouse.Mobile.Services.Api;
using Lighthouse.Mobile.Services.Auth;

namespace Lighthouse.Mobile.ViewModels;

public sealed partial class LoginViewModel(ApiClient api, ITokenStore tokenStore) : ObservableObject
{
    [ObservableProperty]
    private string _email = string.Empty;

    [ObservableProperty]
    private string _password = string.Empty;

    [ObservableProperty]
    private string? _errorMessage;

    [ObservableProperty]
    private bool _isBusy;

    [RelayCommand]
    private async Task LoginAsync()
    {
        if (IsBusy) return;

        ErrorMessage = null;
        IsBusy = true;
        try
        {
            var auth = await api.LoginAsync(Email.Trim(), Password, CancellationToken.None);
            await tokenStore.SaveAsync(auth.AccessToken, auth.RefreshToken);
            ErrorMessage = $"Успішний вхід: {auth.User.Name}"; // тимчасово, замінимо на навігацію пізніше
        }
        catch (ApiException ex) when (ex.StatusCode is System.Net.HttpStatusCode.Unauthorized)
        {
            ErrorMessage = "Невірний email або пароль.";
        }
        catch (ApiException ex) when (ex.StatusCode is System.Net.HttpStatusCode.Locked)
        {
            ErrorMessage = "Забагато спроб входу. Спробуйте пізніше.";
        }
        catch (Exception)
        {
            ErrorMessage = "Немає з'єднання з сервером. Перевірте інтернет і спробуйте ще раз.";
        }
        finally
        {
            IsBusy = false;
        }
    }

    [RelayCommand]
    private Task GoBackAsync() => Shell.Current.GoToAsync("..");
}

using Lighthouse.Core.AI;
using Xunit;

namespace Lighthouse.Core.Tests;

public class LinUcbModelTests
{
    [Fact]
    public void NewModel_ScoreEqualsAlphaTimesNormOfContext()
    {
        var model = new LinUcbModel(3, alpha: 1.0);

        var score = model.Score([1, 0, 0]);

        Assert.Equal(1.0, score, precision: 10);
    }

    [Fact]
    public void Update_WithPositiveReward_RaisesExpectedRewardOnlyForSeenContext()
    {
        var model = new LinUcbModel(3, alpha: 0.0);
        double[] seen = [1, 0, 0];
        double[] unseen = [0, 1, 0];

        for (var i = 0; i < 10; i++)
        {
            model.Update(seen, 1.0);
        }

        Assert.True(model.Score(seen) > 0.9);
        Assert.Equal(0.0, model.Score(unseen), precision: 10);
    }

    [Fact]
    public void Update_ReducesUncertaintyForSeenContext()
    {
        var model = new LinUcbModel(3, alpha: 1.0);
        double[] context = [1, 0, 0];
        var before = model.Uncertainty(context);

        for (var i = 0; i < 10; i++)
        {
            model.Update(context, 0.5);
        }

        Assert.True(model.Uncertainty(context) < before / 2);
    }

    [Fact]
    public void Snapshot_RoundTrip_PreservesScores()
    {
        var model = new LinUcbModel(3, alpha: 0.7);
        model.Update([1, 0.5, 0], 1.0);
        model.Update([0, 1, 0.2], 0.0);

        var restored = LinUcbModel.FromSnapshot(model.ToSnapshot());

        double[] probe = [0.3, 0.6, 0.9];
        Assert.Equal(model.Score(probe), restored.Score(probe), precision: 12);
    }

    [Fact]
    public void RankDescending_PutsBetterContextFirst()
    {
        var model = new LinUcbModel(2, alpha: 0.0);
        for (var i = 0; i < 5; i++)
        {
            model.Update([1, 0], 1.0);
            model.Update([0, 1], 0.0);
        }

        var order = model.RankDescending([[0, 1], [1, 0]]);

        Assert.Equal(new[] { 1, 0 }, order.ToArray());
    }

    [Fact]
    public void Score_WithWrongDimension_Throws()
    {
        var model = new LinUcbModel(3);

        Assert.Throws<ArgumentException>(() => model.Score([1, 2]));
    }
}
