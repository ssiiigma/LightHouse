using FluentValidation;
using Microsoft.AspNetCore.Mvc;

namespace Lighthouse.Api.Controllers;

[ApiController]
public abstract class ApiControllerBase : ControllerBase
{
    /// <summary>Повертає null, якщо модель коректна, інакше — готову відповідь 400 з переліком помилок.</summary>
    protected async Task<ActionResult?> ValidateAsync<T>(IValidator<T> validator, T model, CancellationToken ct)
    {
        var result = await validator.ValidateAsync(model, ct);
        return result.IsValid
            ? null
            : ValidationProblem(new ValidationProblemDetails(result.ToDictionary()));
    }
}

using Lighthouse.Api.Data;
using Lighthouse.Api.Infrastructure;
using Lighthouse.Core.Entities;
using Lighthouse.Shared.Contracts.Auth;
using Microsoft.EntityFrameworkCore;

namespace Lighthouse.Api.Services.Auth;

public sealed class AuthService(LighthouseDbContext db, ITokenService tokens) : IAuthService
{
    private const int MaxFailedAttempts = 5;                            // FR-8
    private static readonly TimeSpan LockoutDuration = TimeSpan.FromMinutes(15);

    public async Task<AuthResponse> RegisterAsync(RegisterRequest request, CancellationToken ct)
    {
        var email = NormalizeEmail(request.Email);

        if (await db.Users.AnyAsync(u => u.Email == email, ct))
        {
            throw new AppException(StatusCodes.Status409Conflict, "Email is already registered.");
        }

        var user = new User
        {
            Name = request.Name.Trim(),
            Email = email,
            PasswordHash = BCrypt.Net.BCrypt.HashPassword(request.Password)
            // TODO (FR-4): надіслати лист підтвердження та виставляти IsEmailConfirmed після переходу за посиланням.
        };
        user.NotificationSettings = new NotificationSettings { UserId = user.Id };

        db.Users.Add(user);
        return await IssueTokensAsync(user, ct); // користувач і refresh-токен зберігаються однією транзакцією
    }

    public async Task<AuthResponse> LoginAsync(LoginRequest request, CancellationToken ct)
    {
        const string invalidCredentials = "Invalid email or password.";
        var email = NormalizeEmail(request.Email);

        var user = await db.Users.FirstOrDefaultAsync(u => u.Email == email && !u.IsDeleted, ct);
        if (user is null || user.PasswordHash is null)
        {
            throw new AppException(StatusCodes.Status401Unauthorized, invalidCredentials);
        }

        if (user.LockoutEndUtc is { } lockoutEnd && lockoutEnd > DateTime.UtcNow)
        {
            throw new AppException(StatusCodes.Status423Locked,
                $"Account is locked until {lockoutEnd:u}. Try again later.");
        }

        if (!BCrypt.Net.BCrypt.Verify(request.Password, user.PasswordHash))
        {
            user.FailedLoginCount++;
            if (user.FailedLoginCount >= MaxFailedAttempts)
            {
                user.LockoutEndUtc = DateTime.UtcNow + LockoutDuration;
                user.FailedLoginCount = 0;
            }

            await db.SaveChangesAsync(ct);
            throw new AppException(StatusCodes.Status401Unauthorized, invalidCredentials);
        }

        user.FailedLoginCount = 0;
        user.LockoutEndUtc = null;
        return await IssueTokensAsync(user, ct);
    }

    public async Task<AuthResponse> RefreshAsync(string refreshToken, CancellationToken ct)
    {
        var hash = tokens.Hash(refreshToken);
        var stored = await db.RefreshTokens
            .Include(t => t.User)
            .FirstOrDefaultAsync(t => t.TokenHash == hash, ct);

        if (stored is null || stored.RevokedAtUtc is not null || stored.ExpiresAtUtc <= DateTime.UtcNow)
        {
            throw new AppException(StatusCodes.Status401Unauthorized, "Invalid refresh token.");
        }

        stored.RevokedAtUtc = DateTime.UtcNow; // ротація: старий токен більше не діє
        return await IssueTokensAsync(stored.User, ct);
    }

    public async Task LogoutAsync(string refreshToken, CancellationToken ct)
    {
        var hash = tokens.Hash(refreshToken);
        var stored = await db.RefreshTokens.FirstOrDefaultAsync(t => t.TokenHash == hash, ct);

        if (stored is not null && stored.RevokedAtUtc is null)
        {
            stored.RevokedAtUtc = DateTime.UtcNow; // FR-13
            await db.SaveChangesAsync(ct);
        }
    }

    private async Task<AuthResponse> IssueTokensAsync(User user, CancellationToken ct)
    {
        var (accessToken, accessExpiresAt) = tokens.CreateAccessToken(user);
        var refresh = tokens.CreateRefreshToken();

        db.RefreshTokens.Add(new RefreshToken
        {
            UserId = user.Id,
            TokenHash = refresh.Hash,
            ExpiresAtUtc = refresh.ExpiresAtUtc
        });

        await db.SaveChangesAsync(ct);

        return new AuthResponse(accessToken, refresh.Token, accessExpiresAt,
            new UserDto(user.Id, user.Name, user.Email));
    }

    private static string NormalizeEmail(string email) => email.Trim().ToLowerInvariant();
}

using System.Security.Claims;
using System.Security.Cryptography;
using System.Text;
using Lighthouse.Api.Infrastructure;
using Lighthouse.Api.Options;
using Lighthouse.Core.Entities;
using Microsoft.Extensions.Options;
using Microsoft.IdentityModel.JsonWebTokens;
using Microsoft.IdentityModel.Tokens;

namespace Lighthouse.Api.Services.Auth;

public sealed class TokenService(IOptions<JwtOptions> options) : ITokenService
{
    private readonly JwtOptions _options = options.Value;

    public (string Token, DateTime ExpiresAtUtc) CreateAccessToken(User user)
    {
        var expires = DateTime.UtcNow.AddMinutes(_options.AccessTokenMinutes);
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_options.Secret));

        var descriptor = new SecurityTokenDescriptor
        {
            Subject = new ClaimsIdentity(new List<Claim>
            {
                new(TokenClaims.Subject, user.Id.ToString()),
                new(TokenClaims.Email, user.Email),
                new(TokenClaims.Name, user.Name)
            }),
            Issuer = _options.Issuer,
            Audience = _options.Audience,
            Expires = expires,
            SigningCredentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha256)
        };

        return (new JsonWebTokenHandler().CreateToken(descriptor), expires);
    }

    public RefreshTokenInfo CreateRefreshToken()
    {
        var token = Convert.ToBase64String(RandomNumberGenerator.GetBytes(64));
        return new RefreshTokenInfo(token, Hash(token), DateTime.UtcNow.AddDays(_options.RefreshTokenDays));
    }

    public string Hash(string token) =>
        Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(token)));
}

        }

using Lighthouse.Api.Data;
using Lighthouse.Api.Infrastructure;
using Lighthouse.Core.Entities;
using Lighthouse.Shared.Contracts.Categories;
using Microsoft.EntityFrameworkCore;

namespace Lighthouse.Api.Services.Categories;

public sealed class CategoryService(LighthouseDbContext db) : ICategoryService
{
    public async Task<IReadOnlyList<CategoryDto>> GetAsync(Guid userId, DateTime? modifiedSinceUtc, CancellationToken ct)
    {
        var query = db.Categories.AsNoTracking().Where(c => c.UserId == userId);

        query = SyncTime.ToUtc(modifiedSinceUtc) is { } since
            ? query.Where(c => c.ServerModifiedAtUtc > since)
            : query.Where(c => !c.IsDeleted);

        var items = await query.OrderBy(c => c.ServerModifiedAtUtc).ToListAsync(ct);
        return items.Select(ToDto).ToList();
    }

    public async Task<CategoryDto> UpsertAsync(Guid userId, Guid id, UpsertCategoryRequest request, CancellationToken ct)
    {
        var category = await db.Categories.FirstOrDefaultAsync(c => c.Id == id, ct);
        if (category is not null && category.UserId != userId)
        {
            throw new AppException(StatusCodes.Status404NotFound, "Category not found.");
        }

        var now = DateTime.UtcNow;
        var editedAt = SyncTime.ClampToNow(request.ClientUpdatedAtUtc, now);

        if (category is null)
        {
            category = new Category { Id = id, UserId = userId, CreatedAtUtc = now };
            db.Categories.Add(category);
        }
        else if (category.UpdatedAtUtc > editedAt)
        {
            return ToDto(category); // «останній запис»
        }

        category.Name = request.Name.Trim();
        category.ColorHex = request.ColorHex;
        category.Icon = request.Icon;
        category.IsDeleted = false;
        category.UpdatedAtUtc = editedAt;
        category.ServerModifiedAtUtc = now;

        await db.SaveChangesAsync(ct);
        return ToDto(category);
    }

    public async Task DeleteAsync(Guid userId, Guid id, CancellationToken ct)
    {
        var category = await db.Categories.FirstOrDefaultAsync(c => c.Id == id, ct);
        if (category is null)
        {
            return; // ідемпотентно
        }

        if (category.UserId != userId)
        {
            throw new AppException(StatusCodes.Status404NotFound, "Category not found.");
        }

        if (category.IsDeleted)
        {
            return;
        }

        var now = DateTime.UtcNow;
        category.IsDeleted = true;
        category.UpdatedAtUtc = now;
        category.ServerModifiedAtUtc = now;

        // FK з NoAction (див. DbContext): відв'язуємо залежні записи вручну, щоб вони потрапили в дельту синхронізації.
        var tasks = await db.Tasks.Where(t => t.UserId == userId && t.CategoryId == id).ToListAsync(ct);
        foreach (var task in tasks)
        {
            task.CategoryId = null;
            task.UpdatedAtUtc = now;
            task.ServerModifiedAtUtc = now;
        }

        var habits = await db.Habits.Where(h => h.UserId == userId && h.CategoryId == id).ToListAsync(ct);
        foreach (var habit in habits)
        {
            habit.CategoryId = null;
            habit.UpdatedAtUtc = now;
            habit.ServerModifiedAtUtc = now;
        }

        await db.SaveChangesAsync(ct);
    }

    private static CategoryDto ToDto(Category c) => new(
        c.Id, c.Name, c.ColorHex, c.Icon, c.UpdatedAtUtc, c.ServerModifiedAtUtc, c.IsDeleted);
}

/// <summary>
/// Зберігання моделі користувача (знімок BanditState з версією). Запис проводиться з перевіркою версії:

public sealed class BanditStore(LighthouseDbContext db)
{
    public async Task<LoadedModel> LoadAsync(Guid userId, CancellationToken ct)
    {
        var state = await db.BanditStates.FirstOrDefaultAsync(s => s.UserId == userId, ct);
        if (state is null)
        {
            return new LoadedModel(NewModel(), null);
        }

        var snapshot = JsonSerializer.Deserialize<LinUcbSnapshot>(state.SnapshotJson);
        var model = snapshot is { Dimension: LinUcbDefaults.Dimension }
            ? LinUcbModel.FromSnapshot(snapshot)
            : NewModel(); // розмірність змінилась між версіями — починаємо з чистої моделі

        return new LoadedModel(model, state);
    }

    /// <summary>Готує запис знімка (без SaveChanges) і повертає нову версію.</summary>
    public int Stage(Guid userId, LoadedModel loaded, DateTime nowUtc)
    {
        var json = JsonSerializer.Serialize(loaded.Model.ToSnapshot());

        if (loaded.State is null)
        {
            db.BanditStates.Add(new BanditState
            {
                UserId = userId,
                Dimension = loaded.Model.Dimension,
                SnapshotJson = json,
                Version = 1,
                UpdatedAtUtc = nowUtc
            });
            return 1;
        }

        loaded.State.Dimension = loaded.Model.Dimension;
        loaded.State.SnapshotJson = json;
        loaded.State.Version++;
        loaded.State.UpdatedAtUtc = nowUtc;
        return loaded.State.Version;
    }

        db.RecommendationFeedbacks.CountAsync(f => f.UserId == userId && f.Reward != null, ct);

    /// <summary>Рекомендація без результату довше цього часу вважається неактуальною й видаляється.</summary>
    private static readonly TimeSpan PendingLifetime = TimeSpan.FromDays(14);

    public async Task<RankingResponse> GetRankingAsync(Guid userId, CancellationToken ct)
    {
        var now = DateTime.UtcNow;
        var zone = await userTime.GetZoneAsync(userId, ct);

        await ResolvePendingAsync(userId, now, ct);

        return await OptimisticRetry.RunAsync(db, async () =>
        {
            var tasks = await db.Tasks.AsNoTracking()
                .Where(t => t.UserId == userId && !t.IsDeleted && t.CompletedAtUtc == null)
                .ToListAsync(ct);

            var history = await historyBuilder.BuildAsync(userId, tasks.Select(t => t.Id).ToList(), zone, now, ct);
            var loaded = await store.LoadAsync(userId, ct);
            var feedbackCount = await store.CountFeedbackAsync(userId, ct);

            var result = prioritizer.Rank(new PrioritizationRequest(
                tasks, history, loaded.Model, feedbackCount, now, zone));

            // Для перших K завдань зберігаємо вектор контексту на момент показу та ключ рекомендації.
            var top = result.Ranked.Take(LinUcbDefaults.TopK).ToList();

            var pending = await db.RecommendationFeedbacks
                .Where(f => f.UserId == userId && f.RespondedAtUtc == null
                            && f.TaskId != null && topIds.Contains(f.TaskId.Value))
                .ToListAsync(ct);

            var keys = new Dictionary<Guid, string>();
            foreach (var ranked in top)
            {
                var row = pending.FirstOrDefault(f => f.TaskId == ranked.TaskId);
                if (row is null)
                {
                    row = new RecommendationFeedback
                    {
                        UserId = userId,
                        TaskId = ranked.TaskId,
                        RecommendationKey = Guid.NewGuid().ToString("N"),
                        CreatedAtUtc = now
                    };
                    db.RecommendationFeedbacks.Add(row);
                }

                row.ContextJson = ContextCodec.Serialize(ranked.Context); // оновлюємо контекст, ключ лишається
                keys[ranked.TaskId] = row.RecommendationKey;
            }

            await db.SaveChangesAsync(ct);

            return new RankingResponse(
                result.Mode.ToString(),
                result.DailyLimit,
                result.Ranked.Select(r => new RankedTaskDto(
                    r.TaskId,
                    r.Position,
                    r.Score,
                    r.IsLayer1,
                    keys.GetValueOrDefault(r.TaskId),

                    r.Explanation.UncertaintyBonus,
                    r.Explanation.IsExploration)).ToList(),

    public Task<FeedbackResponse> SubmitFeedbackAsync(Guid userId, FeedbackRequest request, CancellationToken ct) =>
        OptimisticRetry.RunAsync(db, async () =>
        {
            var feedback = await db.RecommendationFeedbacks
                .FirstOrDefaultAsync(f => f.UserId == userId && f.RecommendationKey == request.RecommendationKey, ct)

            var now = DateTime.UtcNow;

                RecommendationAction.Rejected => (FeedbackKind.Rejected, BehaviorEventType.RecommendationRejected),

    /// Визначає результат показаних, але не підтверджених рекомендацій: ручне відкладення (винагорода за
    /// RewardCalculator.ForManualPostponement) або виконання завдання після показу (1).

    private Task<int> ResolvePendingAsync(Guid userId, DateTime now, CancellationToken ct) =>
        OptimisticRetry.RunAsync(db, async () =>
        {
            var pending = await db.RecommendationFeedbacks
                .Where(f => f.UserId == userId && f.RespondedAtUtc == null && f.TaskId != null)
                .ToListAsync(ct);
            if (pending.Count == 0)
            {
                return 0;
            }

            var taskIds = pending.Select(f => f.TaskId!.Value).Distinct().ToList();
            var tasks = await db.Tasks.AsNoTracking()
                .Where(t => t.UserId == userId && taskIds.Contains(t.Id))
                .ToDictionaryAsync(t => t.Id, ct);
            var postponements = await db.BehaviorEvents.AsNoTracking()
                .Where(e => e.UserId == userId && e.Type == BehaviorEventType.TaskPostponed
                            && e.TaskId != null && taskIds.Contains(e.TaskId.Value))
                .ToListAsync(ct);

            var loaded = await store.LoadAsync(userId, ct);
            var changed = false;

            foreach (var feedback in pending)
            {
                double? reward = null;
                if (tasks.TryGetValue(feedback.TaskId!.Value, out var task))
                {
                    var firstPostponement = postponements
                        .Where(e => e.TaskId == feedback.TaskId && e.OccurredAtUtc >= feedback.CreatedAtUtc)
                        .OrderBy(e => e.OccurredAtUtc)
                        .FirstOrDefault();

                    if (firstPostponement is not null)
                    {
                        reward = RewardCalculator.ForManualPostponement(
                            firstPostponement.OccurredAtUtc, task.DeadlineUtc, task.CompletedAtUtc, now);
                    }
                    else if (task.CompletedAtUtc is { } completedAt && completedAt >= feedback.CreatedAtUtc)
                    {
                        reward = RewardCalculator.ForFeedback(FeedbackKind.Completed);
                    }
                }

                if (reward is { } value)
                {
                    ApplyReward(feedback, value, loaded.Model, now);
                    changed = true;
                }
                else if (now - feedback.CreatedAtUtc > PendingLifetime)
                {
                    db.RecommendationFeedbacks.Remove(feedback);
                    changed = true;
                }
            }

            if (changed)
            {
                store.Stage(userId, loaded, now);
                await db.SaveChangesAsync(ct);
            }

            return 0;
        });

    private static void ApplyReward(RecommendationFeedback feedback, double reward, LinUcbModel model, DateTime now)
    {
        feedback.Reward = reward;
        feedback.IsPositive = reward >= 1.0;
        feedback.RespondedAtUtc = now;

        if (ContextCodec.TryParse(feedback.ContextJson) is { } context)
        {
            model.Update(context, reward); // Шерман–Моррісон
        }
    }

    private static List<ResolutionDto> BuildResolutions(PrioritizationResult result)
    {
        var resolutions = new List<ResolutionDto>();
        if (result.DeferralSuggestions.Count > 0)
        {
            resolutions.Add(new ResolutionDto(ResolutionKind.DeferToNextDay, result.DeferralSuggestions));
        }

        if (result.DoNotFit.Count > 0)
        {
            resolutions.Add(new ResolutionDto(ResolutionKind.DoesNotFit, result.DoNotFit));
        }

        return resolutions;
    }

/// <summary>
/// Агрегує BehaviorEvents за 28 діб у UserHistory (ознаки x6–x9, ліміт доби). Одиниця згладженої частки —
/// «підсумкові» події TaskCompleted / TaskPostponed / TaskSkipped; c — лише TaskCompleted.
/// </summary>
public sealed class UserHistoryBuilder(LighthouseDbContext db)
{
    private const int WindowDays = 28;

    private static readonly BehaviorEventType[] OutcomeTypes =
        [BehaviorEventType.TaskCompleted, BehaviorEventType.TaskPostponed, BehaviorEventType.TaskSkipped];

    public async Task<UserHistory> BuildAsync(
        Guid userId, IReadOnlyList<Guid> openTaskIds, TimeZoneInfo zone, DateTime nowUtc, CancellationToken ct)
    {
        var since = nowUtc.AddDays(-WindowDays);

        var events = await db.BehaviorEvents.AsNoTracking()
            .Where(e => e.UserId == userId && e.OccurredAtUtc >= since && OutcomeTypes.Contains(e.Type))
            .Select(e => new { e.Type, e.TaskId, e.OccurredAtUtc, e.HourOfDay, e.WeekDay })

            ? new Dictionary<Guid, Guid?>()
            : await db.Tasks.AsNoTracking()
                .Where(t => eventTaskIds.Contains(t.Id))
                .ToDictionaryAsync(t => t.Id, t => t.CategoryId, ct);

        var hourDone = new int[UserHistory.HourBuckets];
        var hourTotal = new int[UserHistory.HourBuckets];
        var dayDone = new int[UserHistory.WeekDays];
        var dayTotal = new int[UserHistory.WeekDays];
        var byCategory = new Dictionary<Guid, (int Done, int Total)>();

        foreach (var e in events)
        {
            var completed = e.Type == BehaviorEventType.TaskCompleted;
            var hour = Math.Clamp(e.HourOfDay / 4, 0, UserHistory.HourBuckets - 1);
            var day = Math.Clamp(e.WeekDay, 0, UserHistory.WeekDays - 1);

            hourTotal[hour]++;
            dayTotal[day]++;
            if (completed)
            {
                hourDone[hour]++;
                dayDone[day]++;
            }

            if (e.TaskId is { } taskId && categories.TryGetValue(taskId, out var categoryId) && categoryId is { } cid)
            {
                byCategory.TryGetValue(cid, out var counts);
                byCategory[cid] = (counts.Done + (completed ? 1 : 0), counts.Total + 1);
            }
        }

        var postponeCounts = openTaskIds.Count == 0
            ? new Dictionary<Guid, int>()
            : await db.BehaviorEvents.AsNoTracking()
                .Where(e => e.UserId == userId && e.Type == BehaviorEventType.TaskPostponed
                            && e.TaskId != null && openTaskIds.Contains(e.TaskId.Value))
                .GroupBy(e => e.TaskId!.Value)
                .Select(g => new { Id = g.Key, Count = g.Count() })
                .ToDictionaryAsync(x => x.Id, x => x.Count, ct);

        var perDay = events
            .Where(e => e.Type == BehaviorEventType.TaskCompleted)
            .GroupBy(e => TimeZoneInfo.ConvertTimeFromUtc(e.OccurredAtUtc, zone).Date)
            .Select(g => g.Count())
            .OrderBy(c => c)
            .ToList();

        return new UserHistory
        {
            ByHourBucket = hourDone.Select((d, i) => new OutcomeCount(d, hourTotal[i])).ToList(),
            ByWeekDay = dayDone.Select((d, i) => new OutcomeCount(d, dayTotal[i])).ToList(),
            ByCategory = byCategory.ToDictionary(kv => kv.Key, kv => new OutcomeCount(kv.Value.Done, kv.Value.Total)),
            PostponeCounts = postponeCounts,
            MedianCompletedPerDay = Median(perDay)
        };
    }

    private static double? Median(List<int> sorted)
    {
        if (sorted.Count == 0)
        {
            return null;
        }

        var middle = sorted.Count / 2;
        return sorted.Count % 2 == 1 ? sorted[middle] : (sorted[middle - 1] + sorted[middle]) / 2.0;
    }
}

    {
        var query = db.Tasks.AsNoTracking().Include(t => t.Subtasks).Where(t => t.UserId == userId);

        query = SyncTime.ToUtc(modifiedSinceUtc) is { } since
            ? query.Where(t => t.ServerModifiedAtUtc > since)   // дельта: разом із «надгробками» видалених
            : query.Where(t => !t.IsDeleted);                   // перше завантаження

        var items = await query.OrderBy(t => t.ServerModifiedAtUtc).ToListAsync(ct);
        return items.Select(ToDto).ToList();
    }

    public async Task<TaskDto> UpsertAsync(Guid userId, Guid id, UpsertTaskRequest request, CancellationToken ct)
    {
        if (request.CategoryId is { } categoryId &&
            !await db.Categories.AnyAsync(c => c.Id == categoryId && c.UserId == userId && !c.IsDeleted, ct))
        {
            throw new AppException(StatusCodes.Status400BadRequest, "Category not found.");
        }

        var task = await db.Tasks.Include(t => t.Subtasks).FirstOrDefaultAsync(t => t.Id == id, ct);

        // Чужий запис маскуємо під «не існує», щоб не розкривати наявність ідентифікатора.
        if (task is not null && task.UserId != userId)
        {
            throw new AppException(StatusCodes.Status404NotFound, "Task not found.");
        }

        var now = DateTime.UtcNow;
        var editedAt = SyncTime.ClampToNow(request.ClientUpdatedAtUtc, now);
        var isNew = task is null;

        if (task is null)
        {
            task = new TaskItem { Id = id, UserId = userId, CreatedAtUtc = now };
            db.Tasks.Add(task);
        }
        else if (task.UpdatedAtUtc > editedAt)
        {
            return ToDto(task);
        }

        // Стан «до» — для визначення переходів, що породжують поведінкові події (повторна відправка подій не дублює).
        var previousDeadline = task.DeadlineUtc;
        var wasCompleted = task.CompletedAtUtc is not null;

        task.Title = request.Title.Trim();
        task.Description = request.Description;
        task.DeadlineUtc = SyncTime.ToUtc(request.DeadlineUtc);
        task.Priority = request.Priority;
        task.CategoryId = request.CategoryId;
        task.ReminderAtUtc = SyncTime.ToUtc(request.ReminderAtUtc);
        task.CompletedAtUtc = SyncTime.ToUtc(request.CompletedAtUtc);
        task.EstimatedMinutes = request.EstimatedMinutes;
        task.IsDeleted = false;
        task.UpdatedAtUtc = editedAt;
        task.ServerModifiedAtUtc = now;

        SyncSubtasks(task, request.Subtasks, now);

        if (isNew)
        {
            await behavior.RecordAsync(userId, BehaviorEventType.TaskCreated, editedAt, taskId: id, ct: ct);
        }

        if (!wasCompleted && task.CompletedAtUtc is { } completedAt)
        {
            await behavior.RecordAsync(userId, BehaviorEventType.TaskCompleted,
                SyncTime.ClampToNow(completedAt, now), taskId: id, ct: ct);
        }

        // «Відкладання» = перенесення дедлайну існуючого завдання на пізніший момент.
        if (previousDeadline is { } oldDeadline && task.DeadlineUtc is { } newDeadline && newDeadline > oldDeadline)
        {
            var payload = JsonSerializer.Serialize(new
            {
                PreviousDeadlineUtc = SyncTime.ToUtc(oldDeadline),
                NewDeadlineUtc = newDeadline
            });
            await behavior.RecordAsync(userId, BehaviorEventType.TaskPostponed, editedAt,
                taskId: id, payloadJson: payload, ct: ct);
        }

        await db.SaveChangesAsync(ct);
        return ToDto(task);
    }

    public async Task DeleteAsync(Guid userId, Guid id, CancellationToken ct)
    {
        var task = await db.Tasks.FirstOrDefaultAsync(t => t.Id == id, ct);
        if (task is null)
        {
            return; // видалення ідемпотентне
        }

        if (task.UserId != userId)
        {
            throw new AppException(StatusCodes.Status404NotFound, "Task not found.");
        }

        if (task.IsDeleted)
        {
            return;
        }

        var now = DateTime.UtcNow;
        task.IsDeleted = true; // архів; фізичне видалення через 14 днів — окрема фонова задача (UC-09)
        task.UpdatedAtUtc = now;
        task.ServerModifiedAtUtc = now;

        // Видалення невиконаного завдання вважається його пропуском.
        if (task.CompletedAtUtc is null)
        {
            await behavior.RecordAsync(userId, BehaviorEventType.TaskSkipped, now, taskId: id, ct: ct);
        }

        await db.SaveChangesAsync(ct);
    }

    private void SyncSubtasks(TaskItem task, IReadOnlyList<SubtaskDto> incoming, DateTime now)
    {
        var incomingIds = incoming.Select(s => s.Id).ToHashSet();

        foreach (var removed in task.Subtasks.Where(s => !incomingIds.Contains(s.Id)).ToList())
        {
            db.Subtasks.Remove(removed);
        }

        for (var order = 0; order < incoming.Count; order++)
        {
            var dto = incoming[order];
            var existing = task.Subtasks.FirstOrDefault(s => s.Id == dto.Id);

            if (existing is null)
            {
                // Add() явно, щоб EF не вважав сутність із заданим ключем уже існуючою.
                db.Subtasks.Add(new Subtask
                {
                    Id = dto.Id,
                    TaskItemId = task.Id,
                    Title = dto.Title.Trim(),
                    IsDone = dto.IsDone,
                    SortOrder = order,
                    UpdatedAtUtc = now,
                    ServerModifiedAtUtc = now
                });
            }
            else
            {
                existing.Title = dto.Title.Trim();
                existing.IsDone = dto.IsDone;
                existing.SortOrder = order;
                existing.UpdatedAtUtc = now;
                existing.ServerModifiedAtUtc = now;
            }
        }
    }

    private static TaskDto ToDto(TaskItem t) => new(
        t.Id,
        t.Title,
        t.Description,
        t.DeadlineUtc,
        t.Priority,
        t.CategoryId,
        t.ReminderAtUtc,

        t.EstimatedMinutes,
        t.Subtasks.OrderBy(s => s.SortOrder).Select(s => new SubtaskDto(s.Id, s.Title, s.IsDone)).ToList(),
        t.UpdatedAtUtc,
        t.ServerModifiedAtUtc,
        t.IsDeleted);

public interface ILocalDB

public sealed class LocalDB : ILocalDB
{
    private readonly SemaphoreSlim _gate = new(1, 1);
    private SQLiteAsyncConnection? _connection;

    public async Task<SQLiteAsyncConnection> GetConnectionAsync()
    {
        if (_connection is not null)
        {
            return _connection;
        }

        await _gate.WaitAsync();
        try
        {
            if (_connection is null)
            {
                var path = Path.Combine(FileSystem.AppDataDirectory, "lighthouse.db3");
                var connection = new SQLiteAsyncConnection(path,
                    SQLiteOpenFlags.ReadWrite | SQLiteOpenFlags.Create | SQLiteOpenFlags.SharedCache);

                await connection.EnableWriteAheadLoggingAsync();
                await connection.CreateTableAsync<LocalCategory>();
                await connection.CreateTableAsync<LocalTask>();
                await connection.CreateTableAsync<LocalSubtask>();
                await connection.CreateTableAsync<LocalHabit>();
                await connection.CreateTableAsync<LocalHabitLog>();
                await connection.CreateTableAsync<OutboxItem>();
                await connection.CreateTableAsync<RankingCache>();
                await connection.CreateTableAsync<SyncMeta>();
                _connection = connection;
            }

            return _connection;
        }
        finally
        {
            _gate.Release();
        }
    }

    public async Task ClearAllAsync()
    {
        var db = await GetConnectionAsync();
        await db.RunInTransactionAsync(tx =>
        {
            tx.DeleteAll<LocalSubtask>();
            tx.DeleteAll<LocalTask>();
            tx.DeleteAll<LocalCategory>();
            tx.DeleteAll<LocalHabitLog>();
            tx.DeleteAll<LocalHabit>();
            tx.DeleteAll<OutboxItem>();
            tx.DeleteAll<RankingCache>();
            tx.DeleteAll<SyncMeta>();
        });
    }

/// <summary>
/// Курсор дельта-синхронізації (максимальний ServerModifiedAtUtc отриманих записів). Лежить у SQLite й оновлюється
/// в одній транзакції з даними: втрата чи відкат бази автоматично скидає курсор, тож пропущених записів не буде.
/// </summary>
internal static class SyncCursor
{
    private static string Key(string entityType) => $"cursor.{entityType}";

    public static async Task<DateTime?> LoadAsync(SQLiteAsyncConnection db, string entityType)
    {
        var row = await db.FindAsync<SyncMeta>(Key(entityType));
        return row is not null
               && DateTime.TryParse(row.Value, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind, out var value)
            ? Utc.Ensure(value)
            : null;
    }

    public static void Save(SQLiteConnection tx, string entityType, DateTime value) =>
        tx.InsertOrReplace(new SyncMeta
        {
            Key = Key(entityType),
            Value = Utc.Ensure(value).ToString("O", CultureInfo.InvariantCulture)
        });

    public static Task ResetAsync(SQLiteAsyncConnection db) => db.DeleteAllAsync<SyncMeta>();

    /// <summary>Старе сховище курсора (Preferences) більше не використовується.</summary>
    public static void ClearLegacy()
    {
        foreach (var entityType in EntityTypes.All)

        }
    }

/// <summary>Приховані поради: ідентифікатор поради → до якого моменту (UTC) не показувати.</summary>
internal static class AdviceStore
{
    private const string Key = "advice.suppressed";
    private static readonly object Gate = new();

    public static bool IsSuppressed(string id)
    {
        lock (Gate)
        {
            return Load().TryGetValue(id, out var until) && until > Utc.Now;
        }
    }

    public static void Suppress(string id, TimeSpan duration)
    {
        lock (Gate)
        {
            var map = Load();
            var now = Utc.Now;
            foreach (var expired in map.Where(p => p.Value <= now).Select(p => p.Key).ToList())
            {
                map.Remove(expired);
            }

            map[id] = now + duration;
            Preferences.Default.Set(Key, JsonSerializer.Serialize(map));
        }
    }

    public static void Clear() => Preferences.Default.Remove(Key);

    private static Dictionary<string, DateTime> Load()
    {
        try
        {
            return JsonSerializer.Deserialize<Dictionary<string, DateTime>>(Preferences.Default.Get(Key, "{}")) ?? new();
        }
        catch (JsonException)
        {
            return new();
        }
    }
}

public sealed class AIClientService(
    ILocalDB localDb, IApiClient api, IConnectivity connectivity, ISyncSvc sync,
    LocalizationService loc, ILogger<AIClientService> logger) : IAIClient
{
    private static readonly TimeSpan SuppressionShort = TimeSpan.FromHours(12);
    private static readonly TimeSpan SuppressionLong = TimeSpan.FromDays(7);
    private static readonly TimeSpan KeyLifetime = TimeSpan.FromDays(13);
    private static readonly TimeSpan ServerTimeout = TimeSpan.FromSeconds(4);

    public async Task<IReadOnlyList<AdviceItem>> GetAdviceAsync(CancellationToken ct)
    {
        var db = await localDb.GetConnectionAsync();
        var rows = await db.Table<LocalTask>().Where(t => !t.IsDeleted).ToListAsync();
        var open = rows.Where(t => t.CompletedAtUtc is null).ToDictionary(t => t.Id);
        var completedLocal = rows
            .Where(t => t.CompletedAtUtc is not null)
            .Select(t => ToLocal(t.CompletedAtUtc!.Value))
            .ToList();

        // Сервер бачить лише надіслані зміни: поки на пристрої є непереслані, радимо за локальними даними.
        var hasPending = await db.Table<OutboxItem>().CountAsync() > 0;
        var ranking = hasPending ? null : await TryGetRankingAsync(db, ct);

        var obvious = ObviousChoice(open.Values);
        var items = new List<AdviceItem>();
        var focus = ranking is not null
            ? FocusFromServer(ranking, open, obvious)
            : await FocusOfflineAsync(db, open, obvious);
        if (focus is not null)
        {
            items.Add(focus);
        }

        items.AddRange(RescheduleAdvice(open, completedLocal));
        if (ranking is not null)
        {
            items.AddRange(ResolutionAdvice(ranking, open));
        }

        return items;
    }

    public async Task ResolveAsync(AdviceItem item, bool applied, CancellationToken ct = default)
    {

        if (item.Kind == AdviceKind.Focus
            && item.RecommendationKey is { } key
            && Guid.TryParse(key, out var keyId))
        {
            var payload = JsonSerializer.Serialize(
                new FeedbackRequest(key, applied ? RecommendationAction.Accepted : RecommendationAction.Rejected),
                AppJson.Options);

            var db = await localDb.GetConnectionAsync();
            await db.RunInTransactionAsync(tx =>
                OutboxStore.Enqueue(tx, EntityTypes.Feedback, keyId, OutboxStore.Put, payload));
            sync.RequestSync();
        }

    private async Task<RankingResponse?> TryGetRankingAsync(SQLiteAsyncConnection db, CancellationToken ct)
    {
        if (connectivity.NetworkAccess != NetworkAccess.Internet)
        {
            return null;
        }

        using var limit = CancellationTokenSource.CreateLinkedTokenSource(ct);
        limit.CancelAfter(ServerTimeout); // порада не повинна чекати повільний сервер

        try

            await EnsureTimeZoneAsync(limit.Token);
            var ranking = await api.GetAsync<RankingResponse>("api/recommendations/ranking", limit.Token);
            await CacheAsync(db, ranking);
            return ranking;
        }
        catch (Exception ex) when (ex is not OperationCanceledException || !ct.IsCancellationRequested)
        {
            logger.LogInformation(ex, "Server ranking is unavailable; using the offline heuristic.");
            return null;
        }
    }

    private static async Task CacheAsync(SQLiteAsyncConnection db, RankingResponse ranking)
    {

    /// Рекомендація, що збігається з ним, нічого не додає і не показується.

    private static HashSet<Guid> ObviousChoice(IEnumerable<LocalTask> open)
    {
        var list = open.ToList();
        if (list.Count == 0)
        {
            return [];
        }

        var top = list.Max(t => t.Priority);
        var group = list.Where(t => t.Priority == top).ToList();
        var earliest = group.Min(t => t.DeadlineUtc ?? DateTime.MaxValue);
        return group.Where(t => (t.DeadlineUtc ?? DateTime.MaxValue) == earliest).Select(t => t.Id).ToHashSet();
    }

    private AdviceItem? FocusFromServer(RankingResponse ranking, Dictionary<Guid, LocalTask> open, HashSet<Guid> obvious)
    {
        var top = ranking.Ranked.FirstOrDefault();
        if (top is null || top.RecommendationKey is null || !open.TryGetValue(top.TaskId, out var task))
        {
            return null;
        }

        if (obvious.Contains(top.TaskId) || AdviceStore.IsSuppressed($"focus:{top.TaskId}"))
        {
            return null;
        }

        return BuildFocus(task, top.Reasons.Select(r => r.Code).ToList(), top.IsExploration, top.RecommendationKey);
    }

    /// <summary>Офлайн-евристика; ключ відгуку береться з кешу останнього ранжування, якщо завдання було в топ-3.</summary>
    private async Task<AdviceItem?> FocusOfflineAsync(
        SQLiteAsyncConnection db, Dictionary<Guid, LocalTask> open, HashSet<Guid> obvious)
    {
        var ranked = OfflineRanker.Rank(
            open.Values.Select(t => new OfflineTask(t.Id, t.Priority, Utc.Ensure(t.DeadlineUtc))), Utc.Now);

        var top = ranked.FirstOrDefault();
        if (top is null || obvious.Contains(top.TaskId) || AdviceStore.IsSuppressed($"focus:{top.TaskId}"))
        {
            return null;
        }

        var cache = await db.Table<RankingCache>().ToListAsync();
        var key = cache
            .FirstOrDefault(c => c.TaskId == top.TaskId && Utc.Now - Utc.Ensure(c.ComputedAtUtc) < KeyLifetime)
            ?.RecommendationKey;
        return BuildFocus(open[top.TaskId], top.ReasonCodes, false, key);
    }

    private AdviceItem? BuildFocus(LocalTask task, IReadOnlyList<string> codes, bool exploration, string? key)
    {

        var reasons = codes
            .Where(c => !c.StartsWith("priority_", StringComparison.Ordinal))
            .Select(ReasonText)
            .OfType<string>()
            .ToList();

        if (reasons.Count == 0 && !exploration)
        {
            return null; // нічого, що варто пояснювати
        }

        var detail = reasons.Count > 0 ? loc.Get("Advice_Why", string.Join("; ", reasons)) : string.Empty;
        if (exploration)
        {
            detail = (detail + " " + loc["Advice_Exploration"]).Trim();
        }

        return new AdviceItem
        {
            Id = $"focus:{task.Id}",
            Kind = AdviceKind.Focus,
            Title = loc.Get("Advice_FocusTitle", task.Title),
            Summary = reasons.Count > 0 ? Capitalize(reasons[0]) : loc["Advice_FocusDefault"],
            Detail = detail,
            TaskIds = [task.Id],
            RecommendationKey = key,
            IsActionable = true,
            ApplyText = loc["Btn_Accept"],
            DismissText = loc["Btn_Reject"]
        };
    }

    private IEnumerable<AdviceItem> RescheduleAdvice(Dictionary<Guid, LocalTask> open, List<DateTime> completedLocal)
    {
        var candidates = open.Values
            .Where(t => t.ReminderAtUtc is not null)
            .Select(t => new ScheduleTask(
                t.Id, ToLocal(t.ReminderAtUtc!.Value), t.DeadlineUtc is { } d ? ToLocal(d) : null))
            .ToList();

        foreach (var suggestion in ScheduleAdvisor.Suggest(DateTime.Now, completedLocal, candidates).Take(3))
        {
            var id = $"resched:{suggestion.TaskId}:{suggestion.CurrentLocal.Ticks}";
            if (AdviceStore.IsSuppressed(id))
            {
                continue;
            }

            var task = open[suggestion.TaskId];
            yield return new AdviceItem
            {
                Id = id,
                Kind = AdviceKind.Reschedule,
                Title = loc.Get("Advice_Reschedule_Title", task.Title),
                Summary = loc["Advice_Reschedule_Summary"],
                Detail = loc.Get("Advice_Reschedule_Detail",
                    FormatMoment(suggestion.CurrentLocal), BlockLabel(suggestion.BestBlock), FormatMoment(suggestion.ProposedLocal)),
                TaskIds = [task.Id],
                ProposedReminderUtc = DateTime.SpecifyKind(suggestion.ProposedLocal, DateTimeKind.Local).ToUniversalTime(),
                IsActionable = true,
                ApplyText = loc["Btn_Apply"],
                DismissText = loc["Btn_Dismiss"]
            };
        }
    }

    private IEnumerable<AdviceItem> ResolutionAdvice(RankingResponse ranking, Dictionary<Guid, LocalTask> open)
    {
        foreach (var resolution in ranking.Resolutions)
        {
            var ids = resolution.TaskIds.Where(open.ContainsKey).ToList();

            if (resolution.Kind == ResolutionKind.DeferToNextDay)
            {
                // Переносити можна лише завдання, прив'язані до часу (з дедлайном).
                ids = ids.Where(id => open[id].DeadlineUtc is not null).ToList();
                if (ids.Count == 0 || AdviceStore.IsSuppressed($"defer:{Hash(ids)}"))
                {
                    continue;
                }

                yield return new AdviceItem
                {
                    Id = $"defer:{Hash(ids)}",
                    Kind = AdviceKind.Defer,
                    Title = loc["Advice_Defer_Title"],
                    Summary = TitlesList(ids, open),
                    Detail = loc.Get("Advice_Defer_Detail", ranking.DailyLimit, TitlesList(ids, open)),
                    TaskIds = ids,
                    IsActionable = true,
                    ApplyText = loc["Btn_Apply"],
                    DismissText = loc["Btn_Dismiss"]
                };
            }
            else if (resolution.Kind == ResolutionKind.DoesNotFit && ids.Count > 0
                     && !AdviceStore.IsSuppressed($"fit:{Hash(ids)}"))
            {
                yield return new AdviceItem
                {
                    Id = $"fit:{Hash(ids)}",
                    Kind = AdviceKind.DoesNotFit,
                    Title = loc["Advice_Fit_Title"],
                    Summary = TitlesList(ids, open),
                    Detail = loc.Get("Advice_Fit_Detail", TitlesList(ids, open)),
                    TaskIds = ids,
                    IsActionable = false,
                    DismissText = loc["Btn_Hide"]
                };
            }
        }
    }

    private static string FormatMoment(DateTime local) => local.ToString("ddd d MMM, t", CultureInfo.CurrentCulture);

    private static string BlockLabel(int block) => $"{block * 4:00}:00–{block * 4 + 4:00}:00";

/// <summary>Клієнт REST API: підстановка JWT, автоматичне оновлення токена при 401, помилки у вигляді <see cref="ApiException"/>.</summary>
public interface IApiClient
{
    /// <summary>Виникає, коли refresh-токен відхилено сервером: потрібен повторний вхід.</summary>
    event Action? SessionExpired;

    Task<AuthResponse> RegisterAsync(string name, string email, string password, CancellationToken ct);
    Task<AuthResponse> LoginAsync(string email, string password, CancellationToken ct);
    Task LogoutAsync(string refreshToken, CancellationToken ct);
    Task<TResponse> GetAsync<TResponse>(string path, CancellationToken ct);
    Task<byte[]> GetBytesAsync(string path, CancellationToken ct);
    Task<TResponse> PutAsync<TRequest, TResponse>(string path, TRequest body, CancellationToken ct);
    Task DeleteAsync(string path, CancellationToken ct);
    Task<TResponse> PostAsync<TRequest, TResponse>(string path, TRequest body, CancellationToken ct);
}

public sealed class ApiClient(HttpClient http, ISecureStore secureStore) : IApiClient
{
    // Паралельні запити з простроченим токеном оновлюють його по одному: refresh-токен одноразовий (ротація).
    private readonly SemaphoreSlim _refreshGate = new(1, 1);

    public event Action? SessionExpired;

    public async Task<AuthResponse> RegisterAsync(string name, string email, string password, CancellationToken ct)
    {
        using var response = await http.PostAsJsonAsync("api/auth/register", new RegisterRequest(name, email, password), ct);
        return await ReadAuthResponseAsync(response, ct);
    }

    public async Task<AuthResponse> LoginAsync(string email, string password, CancellationToken ct)
    {
        using var response = await http.PostAsJsonAsync("api/auth/login", new LoginRequest(email, password), ct);
        return await ReadAuthResponseAsync(response, ct);
    }

    public async Task LogoutAsync(string refreshToken, CancellationToken ct)
    {
        using var response = await http.PostAsJsonAsync("api/auth/logout", new LogoutRequest(refreshToken), ct);
        await EnsureSuccessAsync(response, ct);
    }

    public async Task<TResponse> GetAsync<TResponse>(string path, CancellationToken ct)
    {
        using var response = await SendAsync(() => new HttpRequestMessage(HttpMethod.Get, path), ct);
        await EnsureSuccessAsync(response, ct);
        return await ReadBodyAsync<TResponse>(response, ct);
    }

    public async Task<TResponse> PutAsync<TRequest, TResponse>(string path, TRequest body, CancellationToken ct)
    {
        using var response = await SendAsync(
            () => new HttpRequestMessage(HttpMethod.Put, path) { Content = JsonContent.Create(body) }, ct);
        await EnsureSuccessAsync(response, ct);
        return await ReadBodyAsync<TResponse>(response, ct);
    }

    public async Task DeleteAsync(string path, CancellationToken ct)
    {
        using var response = await SendAsync(() => new HttpRequestMessage(HttpMethod.Delete, path), ct);
        await EnsureSuccessAsync(response, ct);
    }

    private async Task<HttpResponseMessage> SendAsync(Func<HttpRequestMessage> requestFactory, CancellationToken ct)
    {
        var (accessToken, _) = await secureStore.GetAsync();
        var response = await SendWithTokenAsync(requestFactory(), accessToken, ct);
        if (response.StatusCode != HttpStatusCode.Unauthorized)
        {
            return response;
        }

        response.Dispose();
        var renewed = await RefreshAsync(accessToken ?? string.Empty, ct)
                      ?? throw new ApiException(HttpStatusCode.Unauthorized, "Session expired.");

        return await SendWithTokenAsync(requestFactory(), renewed, ct);
    }
    
    public async Task<byte[]> GetBytesAsync(string path, CancellationToken ct)
    {
        using var response = await SendAsync(() => new HttpRequestMessage(HttpMethod.Get, path), ct);
        await EnsureSuccessAsync(response, ct);
        return await response.Content.ReadAsByteArrayAsync(ct);
    }

    private async Task<string?> RefreshAsync(string failedAccessToken, CancellationToken ct)
    {
        await _refreshGate.WaitAsync(ct);
        try
        {
            var (accessToken, refreshToken) = await secureStore.GetAsync();

            // Інший запит уже оновив токен, поки ми чекали.
            if (!string.IsNullOrEmpty(accessToken) && accessToken != failedAccessToken)
            {
                return accessToken;
            }

            if (string.IsNullOrEmpty(refreshToken))
            {
                return null;
            }

            using var response = await http.PostAsJsonAsync("api/auth/refresh", new RefreshRequest(refreshToken), ct);
            if (!response.IsSuccessStatusCode)
            {
                // Токени стираємо лише коли сервер явно їх відхилив; збій сервера/мережі не завершує сеанс.
                if (response.StatusCode == HttpStatusCode.Unauthorized)
                {
                    await secureStore.ClearAsync();
                    SessionExpired?.Invoke();
                }

                return null;
            }

            var auth = await ReadBodyAsync<AuthResponse>(response, ct);
            await secureStore.SaveAsync(auth.AccessToken, auth.RefreshToken);
            return auth.AccessToken;
        }
        finally
        {
            _refreshGate.Release();
        }
    }

    private async Task<HttpResponseMessage> SendWithTokenAsync(HttpRequestMessage request, string? token, CancellationToken ct)
    {
        if (!string.IsNullOrEmpty(token))
        {
            request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        }

        return await http.SendAsync(request, ct);
    }

    private static async Task<AuthResponse> ReadAuthResponseAsync(HttpResponseMessage response, CancellationToken ct)

        await EnsureSuccessAsync(response, ct);
        return await ReadBodyAsync<AuthResponse>(response, ct);
    }

    private static async Task<T> ReadBodyAsync<T>(HttpResponseMessage response, CancellationToken ct) =>
        await response.Content.ReadFromJsonAsync<T>(cancellationToken: ct)
        ?? throw new InvalidOperationException("Empty response body.");

    private static async Task EnsureSuccessAsync(HttpResponseMessage response, CancellationToken ct)
    {
        if (!response.IsSuccessStatusCode)
        {
            throw new ApiException(response.StatusCode, await response.Content.ReadAsStringAsync(ct));
        }
    }
    
    public async Task<TResponse> PostAsync<TRequest, TResponse>(string path, TRequest body, CancellationToken ct)
    {
        using var response = await SendAsync(
            () => new HttpRequestMessage(HttpMethod.Post, path) { Content = JsonContent.Create(body) }, ct);
        await EnsureSuccessAsync(response, ct);
        return await ReadBodyAsync<TResponse>(response, ct);
    }
}

public sealed class AuthManager : IAuth
{
    private readonly IApiClient _api;
    private readonly ISecureStore _secureStore;
    private readonly ILocalDB _localDb;
    private readonly ISyncSvc _sync;

    public event Action? SessionExpired;

    public AuthManager(IApiClient api, ISecureStore secureStore, ILocalDB localDb, ISyncSvc sync)
    {
        _api = api;
        _secureStore = secureStore;
        _localDb = localDb;
        _sync = sync;
        _api.SessionExpired += () =>
        {
            _sync.Stop();
            SessionExpired?.Invoke();
        };
    }

    public async Task<bool> IsAuthenticatedAsync()
    {
        var (_, refreshToken) = await _secureStore.GetAsync();
        return !string.IsNullOrEmpty(refreshToken);
    }

    public async Task<AuthResponse> RegisterAsync(string name, string email, string password, CancellationToken ct = default)
    {
        var auth = await _api.RegisterAsync(name, email, password, ct);
        await OnSignedInAsync(auth);
        return auth;
    }

    public async Task<AuthResponse> LoginAsync(string email, string password, CancellationToken ct = default)
    {
        var auth = await _api.LoginAsync(email, password, ct);
        await OnSignedInAsync(auth);
        return auth;
    }

    public async Task LogoutAsync(CancellationToken ct = default)
    {
        _sync.Stop();

        var (_, refreshToken) = await _secureStore.GetAsync();
        if (!string.IsNullOrEmpty(refreshToken))
        {
            try
            {
                await _api.LogoutAsync(refreshToken, ct);
            }
            catch (Exception)
            {
                // Без мережі токен на сервері не відкликається й діє до закінчення терміну (30 діб).
            }
        }

        await _secureStore.ClearAsync();
        await ClearLocalDataAsync();
    }

    public async Task<UserProfile> GetProfileAsync()
    {
        var name = Preferences.Default.Get(PreferenceKeys.UserName, string.Empty);
        var email = Preferences.Default.Get(PreferenceKeys.UserEmail, string.Empty);

        if (name.Length == 0)
        {
            var (accessToken, _) = await _secureStore.GetAsync();
            if (TryReadClaims(accessToken, out var tokenName, out var tokenEmail))
            {
                name = tokenName;
                email = tokenEmail;
                Preferences.Default.Set(PreferenceKeys.UserName, name);
                Preferences.Default.Set(PreferenceKeys.UserEmail, email);
            }
        }

        return new UserProfile(name, email);
    }

    private async Task OnSignedInAsync(AuthResponse auth)
    {
        // Дані іншого облікового запису на пристрої не лишаємо.
        var previousUser = Preferences.Default.Get(PreferenceKeys.UserId, string.Empty);
        if (previousUser != auth.User.Id.ToString())
        {
            await ClearLocalDataAsync();
        }

        Preferences.Default.Set(PreferenceKeys.UserId, auth.User.Id.ToString());
        Preferences.Default.Set(PreferenceKeys.UserName, auth.User.Name);
        Preferences.Default.Set(PreferenceKeys.UserEmail, auth.User.Email);
        await _secureStore.SaveAsync(auth.AccessToken, auth.RefreshToken);
        try
        {
            using var limit = new CancellationTokenSource(TimeSpan.FromSeconds(20));
            await _sync.SyncNowAsync(limit.Token);
        }
        catch (Exception)
        {
            // Помилку покаже екран синхронізації; вхід від цього не залежить.
        }

        _sync.Start();
    }

    private async Task ClearLocalDataAsync()
    {
        await _localDb.ClearAllAsync();
        SyncCursor.ClearLegacy();
        Preferences.Default.Remove(PreferenceKeys.UserId);
        Preferences.Default.Remove(PreferenceKeys.UserName);
        Preferences.Default.Remove(PreferenceKeys.UserEmail);
        AdviceStore.Clear();
        Preferences.Default.Remove(PreferenceKeys.TimeZoneSent);
    }

    /// <summary>Читає claim-и name і email з корисного навантаження JWT (підпис тут не перевіряється: дані лише для відображення).</summary>
    private static bool TryReadClaims(string? jwt, out string name, out string email)
    {
        name = email = string.Empty;
        var parts = jwt?.Split('.');
        if (parts is not { Length: 3 })
        {
            return false;
        }

        try
        {
            var payload = parts[1].Replace('-', '+').Replace('_', '/');
            payload = payload.PadRight(payload.Length + (4 - payload.Length % 4) % 4, '=');
            using var document = JsonDocument.Parse(Convert.FromBase64String(payload));
            name = document.RootElement.TryGetProperty("name", out var n) ? n.GetString() ?? string.Empty : string.Empty;
            email = document.RootElement.TryGetProperty("email", out var e) ? e.GetString() ?? string.Empty : string.Empty;
            return name.Length > 0;
        }
        catch (Exception)
        {
            return false;
        }
    }

        {

                return;

            if (dto.UpdatedAtUtc > Utc.Ensure(row.UpdatedAtUtc))
            {
                ApplyRemote(tx, dto, force: true);
                return;
            }

            row.SyncState = SyncState.Synced;

            tx.Update(row);
        });

    {

        var cursor = await SyncCursor.LoadAsync(db, EntityType);
        var path = cursor is { } since
            ? "api/categories?modifiedSince=" + Uri.EscapeDataString(
                since.AddSeconds(-CursorOverlapSeconds).ToString("O", CultureInfo.InvariantCulture))
            : "api/categories";

        var items = await api.GetAsync<List<CategoryDto>>(path, ct);
        return await PullHelper.ApplyAsync(db, EntityType, items,
            (tx, dto) => ApplyRemote(tx, dto, force: false), d => d.ServerModifiedAtUtc);
    }

            {
                return;
            }

            if (tx.Find<LocalCategory>(item.EntityId) is { } row)
            {
                row.SyncState = SyncState.Failed;
                tx.Update(row);

    {

        if (!force && local is not null)
        {
            if (local.SyncState != SyncState.Synced
                && !SyncRules.ShouldApplyRemote(Utc.Ensure(local.UpdatedAtUtc), dto.UpdatedAtUtc))
            {
                return false;
            }

            if (local.SyncState == SyncState.Synced && !dto.IsDeleted
                && Utc.Ensure(local.ServerModifiedAtUtc) >= dto.ServerModifiedAtUtc)
            {
                return false;
            }
        }

        tx.Table<OutboxItem>().Delete(i => i.EntityType == EntityTypes.Category && i.EntityId == dto.Id);

        if (dto.IsDeleted)
        {
            tx.Delete<LocalCategory>(dto.Id);
            return local is not null;
        }

        {
            Id = dto.Id,
            Name = dto.Name,
            ColorHex = dto.ColorHex,
            Icon = dto.Icon,
            UpdatedAtUtc = dto.UpdatedAtUtc,
            ServerModifiedAtUtc = dto.ServerModifiedAtUtc,

    private static bool ApplyRemote(SQLiteConnection tx, HabitDto dto, bool force)
    {
        var local = tx.Find<LocalHabit>(dto.Id);
        var applyFields = true;

        if (!force && local is not null)
        {
            if (local.SyncState != SyncState.Synced
                && !SyncRules.ShouldApplyRemote(Utc.Ensure(local.UpdatedAtUtc), dto.UpdatedAtUtc))
            {
                if (dto.IsDeleted)
                {
                    return false; // локальна правка новіша за видалення
                }

                applyFields = false; // поля лишаємо локальні, але відмітки з сервера зливаємо
            }
            else if (local.SyncState == SyncState.Synced && !dto.IsDeleted
                     && Utc.Ensure(local.ServerModifiedAtUtc) >= dto.ServerModifiedAtUtc)
            {
                return false; // вже маємо цю версію (запас перед курсором)
            }
        }

        if (dto.IsDeleted)
        {
            tx.Table<OutboxItem>().Delete(i => i.EntityType == EntityTypes.Habit && i.EntityId == dto.Id);
            RemoveHabit(tx, dto.Id);
            return local is not null;
        }

        if (applyFields)
        {
            tx.Table<OutboxItem>().Delete(i => i.EntityType == EntityTypes.Habit && i.EntityId == dto.Id);

            var row = local ?? new LocalHabit { Id = dto.Id };
            row.Title = dto.Title;
            row.Description = dto.Description;
            row.FrequencyType = dto.FrequencyType;
            row.DaysOfWeekMask = dto.DaysOfWeekMask;
            row.TimesPerWeek = dto.TimesPerWeek;
            row.ReminderMinutes = dto.ReminderTime is { } time ? time.Hour * 60 + time.Minute : null;
            row.ColorHex = dto.ColorHex;
            row.Icon = dto.Icon;
            row.UpdatedAtUtc = dto.UpdatedAtUtc;
            row.ServerModifiedAtUtc = dto.ServerModifiedAtUtc;
            row.IsDeleted = false;
            row.SyncState = SyncState.Synced;
            tx.InsertOrReplace(row);
        }

        MergeLogs(tx, dto);
        return true;
    }

    /// <summary>Відмітки сервера зливаються з локальними: ще не надіслані локальні зміни не втрачаються.</summary>
    private static void MergeLogs(SQLiteConnection tx, HabitDto dto)
    {
        var serverDates = dto.Logs.Select(l => HabitKeys.Format(l.Date)).ToHashSet();

        foreach (var log in tx.Table<LocalHabitLog>().Where(l => l.HabitId == dto.Id).ToList())
        {
            if (log.SyncState == SyncState.Synced && !serverDates.Contains(log.Date))
            {
                tx.Delete(log); // знято на іншому пристрої
            }
        }

        foreach (var serverLog in dto.Logs)
        {
            var date = HabitKeys.Format(serverLog.Date);
            var id = HabitKeys.LogId(dto.Id, date);
            if (tx.Find<LocalHabitLog>(id) is not null)
            {
                continue;
            }

            // Якщо відмітку щойно зняли локально й видалення ще в черзі, не відновлюємо її.
            var pendingDelete = tx.Table<OutboxItem>().Count(i =>
                i.EntityType == EntityTypes.HabitLog && i.EntityId == id && i.Operation == OutboxStore.Delete) > 0;
            if (pendingDelete)
            {
                continue;
            }

            tx.Insert(new LocalHabitLog
            {
                Id = id,
                HabitId = dto.Id,
                Date = date,
                CompletedAtUtc = serverLog.CompletedAtUtc,
                SyncState = SyncState.Synced
            });
        }
    }

    private static void RemoveHabit(SQLiteConnection tx, Guid id)
    {
        tx.Table<LocalHabitLog>().Delete(l => l.HabitId == id);
        tx.Delete<LocalHabit>(id);

        // Операції з відмітками видаленої звички більше не потрібні.
        foreach (var op in tx.Table<OutboxItem>().Where(i => i.EntityType == EntityTypes.HabitLog).ToList())
        {
            if (HabitKeys.ReadPayload(op.PayloadJson)?.HabitId == id)
            {
                tx.Delete(op);
            }
        }
    }

namespace Lighthouse.Mobile.Services.Sync;

/// <summary>Завантаження частково вдалося: застосовано <see cref="Applied"/> записів, інші відкладено до наступної спроби.</summary>
public sealed class SyncPullException(string message, int applied) : Exception(message)
{
    public int Applied { get; } = applied;
}

internal static class PullHelper
{
    /// <summary>
    /// Застосовує кожен запис сервера в окремій транзакції. Помилка одного запису не зупиняє решту; курсор просувається
    /// лише до останнього успішного запису перед першим збоєм (збійний запис буде отримано знову).
    /// </summary>
    public static async Task<int> ApplyAsync<TDto>(
        SQLiteAsyncConnection db, string entityType, IReadOnlyList<TDto> items,
        Func<SQLiteConnection, TDto, bool> apply, Func<TDto, DateTime> serverModified)
    {
        var applied = 0;
        var failed = false;
        string? firstError = null;
        DateTime? safeCursor = null;

        foreach (var dto in items.OrderBy(serverModified))
        {
            try
            {
                var changed = false;
                await db.RunInTransactionAsync(tx => changed = apply(tx, dto));
                if (changed)
                {
                    applied++;
                }

                if (!failed)
                {
                    safeCursor = serverModified(dto);
                }
            }
            catch (Exception ex)
            {
                failed = true;
                firstError ??= $"{ex.GetType().Name}: {ex.Message}";
            }
        }

        if (safeCursor is { } cursor)
        {
            await db.RunInTransactionAsync(tx => SyncCursor.Save(tx, entityType, cursor));
        }

        if (firstError is not null)
        {
            var text = firstError.Length <= 140 ? firstError : firstError[..140];
            throw new SyncPullException($"{entityType}: {text}", applied);
        }

        return applied;
    }
}

using System.Text.Json;
using Lighthouse.Mobile.Data;
using Lighthouse.Mobile.Infrastructure;
using Lighthouse.Mobile.Services.Api;
using Microsoft.Extensions.Logging;

namespace Lighthouse.Mobile.Services.Sync;

public sealed class SyncService : ISyncSvc
{
    private const string LastOkKey = "sync.last_ok";
    private const string LastErrorKey = "sync.last_error";
    private const int InitialRetrySeconds = 10;
    private const int MaxRetrySeconds = 300;
    private static readonly TimeSpan Debounce = TimeSpan.FromSeconds(1);

    private readonly ILocalDB _localDb;
    private readonly IReadOnlyList<IEntitySyncHandler> _handlers;
    private readonly IConnectivity _connectivity;
    private readonly ILogger<SyncService> _logger;
    private readonly SemaphoreSlim _gate = new(1, 1);

    private CancellationTokenSource? _scheduled;
    private int _retrySeconds = InitialRetrySeconds;

        public async Task<bool> SyncNowAsync(CancellationToken ct = default)
    {
        await _gate.WaitAsync(ct);
        try
        {
            if (_connectivity.NetworkAccess != NetworkAccess.Internet)
            {
                SetStatus(SyncStatus.Offline);
                return false;
            }

            SetStatus(SyncStatus.Syncing);

            var changed = await PushAllAsync(ct);

            // Кожен тип завантажується окремо: збій одного (наприклад, звичок) не блокує решту (завдання).
            var errors = new List<string>();
            foreach (var handler in _handlers)
            {
                try
                {
                    changed += await handler.PullAsync(ct);
                }
                catch (SyncPullException ex)
                {
                    changed += ex.Applied;
                    errors.Add(ex.Message);
                    _logger.LogWarning("Pull of {Type} partially failed: {Message}", handler.EntityType, ex.Message);
                }
                catch (ApiException ex) when (ex.StatusCode == HttpStatusCode.Unauthorized)
                {
                    throw;
                }
                catch (OperationCanceledException) when (ct.IsCancellationRequested)
                {
                    throw;
                }
                catch (Exception ex)
                {
                    errors.Add($"{handler.EntityType}: {Trim(ex.Message)}");
                    _logger.LogWarning(ex, "Pull of {Type} failed.", handler.EntityType);
                }
            }

            if (changed > 0)
            {
                DataChanged?.Invoke();
            }

            if (errors.Count > 0)
            {
                RememberError(string.Join(" | ", errors));
                SetStatus(SyncStatus.Error);
                ScheduleRetry();
                return false;
            }

            _retrySeconds = InitialRetrySeconds;
            Preferences.Default.Set(LastOkKey, DateTime.UtcNow.Ticks.ToString());
            Preferences.Default.Remove(LastErrorKey);
            SetStatus(SyncStatus.Idle);
            return true;
        }
        catch (OperationCanceledException) when (ct.IsCancellationRequested)
        {
            SetStatus(SyncStatus.Idle);
            return false;
        }
        catch (ApiException ex) when (ex.StatusCode == HttpStatusCode.Unauthorized)
        {
            _logger.LogWarning("Synchronization stopped: the session has expired.");
            RememberError("Session expired");
            SetStatus(SyncStatus.Error);
            return false;
        }
        catch (Exception ex)
        {
            _logger.LogWarning(ex, "Synchronization failed; will retry in {Seconds} s.", _retrySeconds);
            RememberError(ex.Message);
            SetStatus(_connectivity.NetworkAccess == NetworkAccess.Internet ? SyncStatus.Error : SyncStatus.Offline);
            ScheduleRetry();
            return false;
        }
        finally
        {
            _gate.Release();
        }
    }

    private async Task<int> PushAllAsync(CancellationToken ct)
    {
        var db = await _localDb.GetConnectionAsync();
        var pushed = 0;

        while (true)
        {
            ct.ThrowIfCancellationRequested();

            // Порядок: категорії, звички, завдання, відмітки, відгуки; всередині типу — за часом створення.
            var queue = await db.Table<OutboxItem>().Where(i => i.Attempts < OutboxStore.MaxAttempts).ToListAsync();
            var item = queue
                .OrderBy(i => _handlers.FirstOrDefault(h => h.EntityType == i.EntityType)?.PullOrder ?? int.MaxValue)
                .ThenBy(i => i.Id)
                .FirstOrDefault();
            if (item is null)
            {
                return pushed;
            }

            var handler = _handlers.FirstOrDefault(h => h.EntityType == item.EntityType);
            if (handler is null)
            {
                await db.DeleteAsync(item);
                continue;
            }

            try
            {
                await handler.PushAsync(item, ct);
                await db.Table<OutboxItem>().DeleteAsync(i => i.Id == item.Id);
                pushed++;
            }
            catch (ApiException ex) when (IsPermanent(ex))
            {
                // Сервер відхилив саме цю зміну (400, 404, 409...): не блокуємо чергу, зміна лишається на пристрої.
                _logger.LogWarning("Server rejected {Type} {Id}: {Status}.", item.EntityType, item.EntityId, ex.StatusCode);
                await ParkAsync(db, handler, item, $"{(int)ex.StatusCode}: {Trim(ex.Message)}");
            }
            catch (ApiException ex) when ((int)ex.StatusCode >= 500)
            {
                item.Attempts++;
                item.LastError = Trim(ex.Message);
                await db.UpdateAsync(item);
                throw;
            }
            catch (Exception ex) when (IsPoison(ex))
            {
                // «Отруйний» елемент (пошкоджені дані черги, помилка обробки): його не вдасться надіслати ніколи,
                // тому він відкладається, а решта черги продовжує надсилатись.
                _logger.LogError(ex, "Outbox item {Type} {Id} cannot be processed.", item.EntityType, item.EntityId);
                await ParkAsync(db, handler, item, $"{ex.GetType().Name}: {Trim(ex.Message)}");
            }
        }
    }

    private static async Task ParkAsync(
        SQLite.SQLiteAsyncConnection db, IEntitySyncHandler handler, OutboxItem item, string error)
    {
        item.Attempts = OutboxStore.MaxAttempts;
        item.LastError = error;
        await db.UpdateAsync(item);
        await handler.MarkFailedAsync(item);
    }

    /// <summary>Помилки 4xx, повторення яких нічого не змінить (крім 401 — потрібен вхід, 408 і 429 — тимчасові).</summary>
    private static bool IsPermanent(ApiException ex) =>
        (int)ex.StatusCode is >= 400 and < 500
        && ex.StatusCode is not (HttpStatusCode.Unauthorized or HttpStatusCode.RequestTimeout or (HttpStatusCode)429);

    private static bool IsPoison(Exception ex) =>
        ex is JsonException or InvalidOperationException or ArgumentException or FormatException
            or NullReferenceException or KeyNotFoundException;

    /// <summary>При запуску й за командою користувача відкладеним змінам даємо ще одну спробу.</summary>
    private async Task ResetParkedAsync()
    {
        var db = await _localDb.GetConnectionAsync();
        var parked = await db.Table<OutboxItem>().Where(i => i.Attempts >= OutboxStore.MaxAttempts).ToListAsync();
        foreach (var item in parked)
        {
            item.Attempts = 0;
            await db.UpdateAsync(item);
        }
    }

    private void OnConnectivityChanged(object? sender, ConnectivityChangedEventArgs e)
    {
        if (e.NetworkAccess == NetworkAccess.Internet)
        {
            RequestSync();
        }
        else
        {
            SetStatus(SyncStatus.Offline);
        }
    }

    private void ScheduleRetry()
    {
        Schedule(TimeSpan.FromSeconds(_retrySeconds));
        _retrySeconds = Math.Min(_retrySeconds * 2, MaxRetrySeconds);
    }

    private void Schedule(TimeSpan delay)
    {
        var cts = new CancellationTokenSource();
        var previous = Interlocked.Exchange(ref _scheduled, cts);
        previous?.Cancel();

        _ = Task.Run(async () =>
        {
            try
            {
                await Task.Delay(delay, cts.Token);
                await SyncNowAsync(cts.Token);
            }
            catch (OperationCanceledException)
            {
                // замінено новішим запитом
            }
        });
    }

    private void CancelScheduled() => Interlocked.Exchange(ref _scheduled, null)?.Cancel();

    private static void RememberError(string message) => Preferences.Default.Set(LastErrorKey, Trim(message));

    private static string Trim(string text) => text.Length <= 160 ? text : text[..160];

    private void SetStatus(SyncStatus status)
    {
        if (Status == status)
        {
            return;
        }

        Status = status;
        StatusChanged?.Invoke(status);
    }
    
    public async Task<bool> FullResyncAsync()
    {
        var db = await _localDb.GetConnectionAsync();
        await SyncCursor.ResetAsync(db);
        return await SyncNowAsync();
    }
}

    /// <summary>Відповідь сервера на PUT: запис синхронізовано, якщо на пристрої немає новіших незбережених змін.</summary>
    private static void Acknowledge(SQLiteConnection tx, OutboxItem item, TaskDto dto)
    {
        var row = tx.Find<LocalTask>(dto.Id);
        if (row is null)
        {
            return;
        }

        if (OutboxStore.HasOtherPending(tx, item))
        {
            return; // користувач змінив запис, поки йшов запит: нова версія ще буде надіслана
        }

        if (dto.UpdatedAtUtc > Utc.Ensure(row.UpdatedAtUtc))
        {
            ApplyRemote(tx, dto, force: true); // на сервері була новіша правка («останній запис»)
            return;
        }

        row.SyncState = SyncState.Synced;
        row.ServerModifiedAtUtc = dto.ServerModifiedAtUtc;
        tx.Update(row);
    }

    /// <summary>Застосовує запис сервера до локальної бази за правилом <see cref="SyncRules"/>.</summary>
    private static bool ApplyRemote(SQLiteConnection tx, TaskDto dto, bool force)
    {
        var local = tx.Find<LocalTask>(dto.Id);

        if (!force && local is not null)
        {
            if (local.SyncState != SyncState.Synced
                && !SyncRules.ShouldApplyRemote(Utc.Ensure(local.UpdatedAtUtc), dto.UpdatedAtUtc))
            {
                return false; // локальна правка новіша — зберігаємо її
            }

            if (local.SyncState == SyncState.Synced && !dto.IsDeleted
                && Utc.Ensure(local.ServerModifiedAtUtc) >= dto.ServerModifiedAtUtc)
            {
                return false; // вже маємо цю версію (запас перед курсором)
            }
        }

        // Версія сервера перемагає: незавершені локальні зміни цього об'єкта скасовуються.
        tx.Table<OutboxItem>().Delete(i => i.EntityType == EntityTypes.Task && i.EntityId == dto.Id);

        if (dto.IsDeleted)
        {
            RemoveTask(tx, dto.Id);
            return local is not null;
        }

        var row = local ?? new LocalTask { Id = dto.Id };
        row.Title = dto.Title;
        row.Description = dto.Description;
        row.DeadlineUtc = dto.DeadlineUtc;
        row.Priority = dto.Priority;
        row.CategoryId = dto.CategoryId;
        row.ReminderAtUtc = dto.ReminderAtUtc;
        row.CompletedAtUtc = dto.CompletedAtUtc;
        row.EstimatedMinutes = dto.EstimatedMinutes;
        row.UpdatedAtUtc = dto.UpdatedAtUtc;
        row.ServerModifiedAtUtc = dto.ServerModifiedAtUtc;
        row.IsDeleted = false;
        row.SyncState = SyncState.Synced;
        tx.InsertOrReplace(row);

        tx.Table<LocalSubtask>().Delete(s => s.TaskId == dto.Id);
        for (var order = 0; order < dto.Subtasks.Count; order++)
        {
            var subtask = dto.Subtasks[order];
            tx.Insert(new LocalSubtask
            {
                Id = subtask.Id,
                TaskId = dto.Id,
                Title = subtask.Title,
                IsDone = subtask.IsDone,
                SortOrder = order
            });
        }

        return true;
    }

    private static void RemoveTask(SQLiteConnection tx, Guid id)
    {
        tx.Table<LocalSubtask>().Delete(s => s.TaskId == id);
        tx.Delete<LocalTask>(id);
    }

        var now = Utc.Now;
        var db = await localDb.GetConnectionAsync();
        await db.RunInTransactionAsync(tx =>
        {

            row.UpdatedAtUtc = now;

        });

    public async Task RecoverAsync()
    {
        var db = await localDb.GetConnectionAsync();
        await db.RunInTransactionAsync(tx =>
        {
            foreach (var row in tx.Table<LocalTask>().Where(t => t.IsDeleted).ToList())
            {
                if (_pendingDeletes.ContainsKey(row.Id))
                {
                    continue;
                }

                var id = row.Id;
                var hasDelete = tx.Table<OutboxItem>().Count(i =>
                    i.EntityType == EntityTypes.Task && i.EntityId == id && i.Operation == OutboxStore.Delete) > 0;
                if (!hasDelete)
                {
                    OutboxStore.Enqueue(tx, EntityTypes.Task, id, OutboxStore.Delete, null);
                }
            }
        });
    }

    private async Task CommitDeleteAfterWindowAsync(Guid id, CancellationTokenSource cts)
    {
        try
        {
            await Task.Delay(UndoWindow, cts.Token);
        }
        catch (OperationCanceledException)
        {
            return; // скасовано користувачем або замінено новим видаленням
        }

        _pendingDeletes.TryRemove(new KeyValuePair<Guid, CancellationTokenSource>(id, cts));

        var db = await localDb.GetConnectionAsync();
        await db.RunInTransactionAsync(tx =>
        {
            if (tx.Find<LocalTask>(id) is { IsDeleted: true })
            {
                OutboxStore.Enqueue(tx, EntityTypes.Task, id, OutboxStore.Delete, null);
            }
        });

        sync.RequestSync();
    }

    private void Notify()
    {
        Changed?.Invoke();
        sync.RequestSync();
    }

            Utc.Ensure(row.UpdatedAtUtc)), AppJson.Options);
